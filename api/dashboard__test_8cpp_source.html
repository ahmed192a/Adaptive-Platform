25std::pair<int, std::string> HttpCall(unsigned short port, std::string const &method, std::string const &target,
26 std::string const &body = {}) {
27 int fd = ::socket(AF_INET, SOCK_STREAM, 0);
28 sockaddr_in a{};
29 a.sin_family = AF_INET;
30 a.sin_port = htons(port);
31 ::inet_pton(AF_INET, "127.0.0.1", &a.sin_addr);
32 if (::connect(fd, reinterpret_cast<sockaddr *>(&a), sizeof a) != 0) return {0, ""};
33 std::string req = method + " " + target + " HTTP/1.1\r\nHost: x\r\nContent-Length: " + std::to_string(body.size()) +
34 "\r\n\r\n" + body;
35 // the server may legitimately answer (and close) before a huge body is fully sent
36 for (std::size_t off = 0; off < req.size();) {
37 ssize_t w = ::send(fd, req.data() + off, req.size() - off, MSG_NOSIGNAL);
38 if (w <= 0) break;
39 off += static_cast<std::size_t>(w);
40 }
41 std::string resp;
42 char buf[4096];
43 ssize_t n;
44 while ((n = ::recv(fd, buf, sizeof buf, 0)) > 0) resp.append(buf, static_cast<std::size_t>(n));
45 ::close(fd);
46 int status = std::stoi(resp.substr(9, 3));
47 return {status, resp.substr(resp.find("\r\n\r\n") + 4)};
48}
