33Result<nlohmann::json> Parse(std::string const &text) {
34 nlohmann::json env = nlohmann::json::parse(text, nullptr, /*allow_exceptions=*/false);
35 if (env.is_discarded() || !env.is_object() || !env.contains("data") || !env["data"].is_object() ||
36 !env.contains("crc32") || !env["crc32"].is_number_unsigned()) {

74Result<void> KeyValueStorage::Load() {
75 std::lock_guard<std::mutex> lock(mMutex);
76 auto primary = LoadFile(Primary(mConfig));
77 if (primary.HasValue()) {
78 mData = mCommitted = primary.Value();
79 return Result<void>();

81 auto backup = LoadFile(Backup(mConfig));

83 mData = mCommitted = backup.Value();

87 PerErrc::kFileNotFound);

89 PerErrc::kFileNotFound);
90 if (primaryMissing && backupMissing) { // brand-new storage
91 mData = mCommitted = mConfig.initialValues;
92 return Result<void>();
93 }
