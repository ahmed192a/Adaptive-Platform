112 TransitionResult RequestTransition(std::string const &to) {
113 std::vector<Action> actions;
114 std::string from;
115 std::uint64_t myGen;
116 {
117 std::lock_guard<std::mutex> g(m_);
118 if (!offered_) return TransitionErrc::kTransitionNotAllowed;
119 if (recovery_) return TransitionErrc::kRecoveryTransitionOngoing;
120 if (!states_.count(to)) return TransitionErrc::kInvalidValue;
121 auto it = transitions_.find({current_, to});
122 if (it == transitions_.end()) it = transitions_.find({"*", to});
123 if (it == transitions_.end()) return TransitionErrc::kTransitionNotAllowed;
124 actions = it->second;
125 from = current_;
126 myGen = ++generation_; // invalidates every older request still running
127 }
128 for (auto const &a : actions) {
129 if (generation_.load() != myGen) return TransitionErrc::kOperationCanceled;
130 if (a.run && !a.run()) {
131 std::lock_guard<std::mutex> g(m_);
132 return generation_ == myGen ? TransitionResult(TransitionErrc::kTransitionFailed)
133 : TransitionResult(TransitionErrc::kOperationCanceled);
134 }
135 }
136 Notifier notify;
137 {
138 std::lock_guard<std::mutex> g(m_);
139 if (generation_ != myGen) return TransitionErrc::kOperationCanceled;
140 current_ = to;
141 notify = notifier_;
142 }
143 if (notify) notify(from, to);
144 return std::nullopt;
145 }
146
147 private:
148 std::string name_;
149 mutable std::mutex m_;
150 std::set<std::string> states_;
151 std::string current_;
152 std::map<std::pair<std::string, std::string>, ActionList> transitions_;
153 Notifier notifier_;
154 bool offered_{true};
155 std::atomic<bool> recovery_{false};
156 std::atomic<std::uint64_t> generation_{0};
157};
