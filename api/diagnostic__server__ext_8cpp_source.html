23std::uint16_t Be16(ByteVector const &v, std::size_t i) {

26void PutBe16(ByteVector &v, std::uint16_t x) {
27 v.push_back(static_cast<std::uint8_t>(x >> 8));
28 v.push_back(static_cast<std::uint8_t>(x & 0xFF));

31 v.push_back(static_cast<std::uint8_t>(dtc >> 16));

36template <class F>
37auto RetryPending(std::uint8_t sid, DiagnosticServer::ResponseSink const &sink, unsigned maxRetries, F &&call) {
38 auto r = call();
39 for (unsigned n = 0; !r.HasValue() && r.Error() == Nrc::kResponsePending && sink && n < maxRetries; ++n) {
40 sink(uds::NegativeResponse(sid, Nrc::kResponsePending));
41 r = call();
42 }
43 return r;
44}

55bool ParseAddressAndLength(ByteVector const &req, std::size_t &pos, std::uint64_t &address, std::uint64_t &length) {
56 if (pos >= req.size()) return false;
57 std::size_t const nAddr = req[pos] & 0x0FU;
58 std::size_t const nLen = static_cast<std::size_t>(req[pos] >> 4U);
59 if (nAddr < 1 || nAddr > 8 || nLen < 1 || nLen > 8 || pos + 1 + nAddr + nLen > req.size()) return false;
60 address = ReadUint(req, pos + 1, nAddr);
61 length = ReadUint(req, pos + 1 + nAddr, nLen);
62 pos += 1 + nAddr + nLen;
63 return true;
64}

223std::optional<ByteVector> DiagnosticServer::HandleReadDtc(ByteVector const &req) {
224 constexpr auto sid = uds::kReadDtcInformation;
225 if (req.size() < 2) return uds::NegativeResponse(sid, Nrc::kIncorrectMessageLengthOrInvalidFormat);
226 std::uint8_t const sub = req[1] & 0x7F;
227 bool const suppress = (req[1] & uds::kSuppressPositiveResponseBit) != 0;
228 switch (sub) {
229 case 0x01: case 0x02: case 0x03: case 0x04: case 0x06: case 0x0A: break;
230 default: return uds::NegativeResponse(sid, Nrc::kSubFunctionNotSupported);
231 }
232 std::size_t const expected = (sub == 0x0A || sub == 0x03) ? 2U : (sub == 0x01 || sub == 0x02) ? 3U : 6U;
233 if (req.size() != expected) return uds::NegativeResponse(sid, Nrc::kIncorrectMessageLengthOrInvalidFormat);
234
235 ByteVector payload{sub};
236 std::lock_guard<std::mutex> lk(mutex_);
237 if (sub == 0x03) { // reportDTCSnapshotIdentification: no availability mask
238 for (auto const &[code, rec] : dtcs_)
239 for (auto const &[num, enc] : rec.snapshots) {
240 (void)enc;
241 PutDtc(payload, code);
242 payload.push_back(num);
243 }
244 } else if (sub == 0x04 || sub == 0x06) {
245 std::uint32_t const code = (static_cast<std::uint32_t>(req[2]) << 16) | (static_cast<std::uint32_t>(req[3]) << 8) | req[4];
246 std::uint8_t const recNum = req[5];
247 auto it = dtcs_.find(code);
248 if (it == dtcs_.end()) return uds::NegativeResponse(sid, Nrc::kRequestOutOfRange);
249 payload.push_back(dtc::kStatusAvailabilityMask);
250 PutDtc(payload, code);
251 payload.push_back(static_cast<std::uint8_t>(it->second.status & dtc::kStatusAvailabilityMask));
252 if (sub == 0x04) {
253 for (auto const &[num, enc] : it->second.snapshots) {
254 if (recNum != 0xFF && recNum != num) continue;
255 payload.push_back(num);
256 payload.insert(payload.end(), enc.begin(), enc.end());
257 }
258 } else {
259 auto ext = it->second.extendedData;
260 ext[0x01] = ByteVector{it->second.occurrences}; // occurrence counter is synthesised
261 for (auto const &[num, data] : ext) {
262 if (recNum < 0xFE && recNum != num) continue;
263 payload.push_back(num);
264 payload.insert(payload.end(), data.begin(), data.end());
265 }
266 }
267 } else {
268 std::uint8_t const mask = (sub == 0x0A) ? 0xFF : req[2];
269 payload.push_back(dtc::kStatusAvailabilityMask);
270 if (sub == 0x01) {
271 std::uint16_t n = 0;
272 for (auto const &[code, rec] : dtcs_) {
273 (void)code;
274 if ((rec.status & mask & dtc::kStatusAvailabilityMask) != 0) ++n;
275 }
276 payload.push_back(0x01); // DTCFormatIdentifier: ISO 14229-1
277 PutBe16(payload, n);
278 } else {
279 for (auto const &[code, rec] : dtcs_) {
280 if (sub == 0x02 && (rec.status & mask & dtc::kStatusAvailabilityMask) == 0) continue;
281 PutDtc(payload, code);
282 payload.push_back(static_cast<std::uint8_t>(rec.status & dtc::kStatusAvailabilityMask));
283 }
284 }
285 }
286 if (suppress) return std::nullopt;
287 return Positive(sid, std::move(payload));
288}

300std::optional<ByteVector> DiagnosticServer::DispatchPhysical(ByteVector const &req) {
301 if (req.empty()) return std::nullopt;
302 try {
303 std::uint8_t const sid = req[0];
304 ServiceHandler custom;
305 ResponseSink sink;
306 {
307 std::lock_guard<std::mutex> lk(mutex_);
308 auto const now = cfg_.clock();
309 ExpireSessionLocked(now);
310 lastActivity_ = now; // any request keeps the session alive
311 sink = pendingSink_;
312 auto it = services_.find(sid);
313 if (it != services_.end()) {
314 if (!InSession(it->second.sessions))
315 return uds::NegativeResponse(sid, Nrc::kServiceNotSupportedInActiveSession);
316 custom = it->second.handler;
317 } else if (auto b = builtinSessions_.find(sid); b != builtinSessions_.end() && !InSession(b->second)) {
318 return uds::NegativeResponse(sid, Nrc::kServiceNotSupportedInActiveSession);
319 }
320 }
321 if (custom) {
322 ByteVector const args(req.begin() + 1, req.end());
323 auto r = RetryPending(sid, sink, cfg_.maxPendingRetries, [&] { return custom(args); });
324 if (!r.HasValue()) return uds::NegativeResponse(sid, r.Error());
325 return Positive(sid, std::move(r).Value());
326 }
327 switch (sid) {
328 case uds::kDiagnosticSessionControl: return HandleSessionControl(req);
329 case uds::kTesterPresent: return HandleTesterPresent(req);
330 case uds::kEcuReset: return HandleEcuReset(req);
331 case uds::kReadDataByIdentifier: return HandleRead(req);
332 case uds::kWriteDataByIdentifier: return HandleWrite(req);
333 case uds::kSecurityAccess: return HandleSecurityAccess(req);
334 case uds::kRoutineControl: return HandleRoutineControl(req);
335 case uds::kClearDiagnosticInformation: return HandleClearDtc(req);
336 case uds::kReadDtcInformation: return HandleReadDtc(req);
337 case uds::kControlDtcSetting: return HandleControlDtcSetting(req);
338 case uds::kCommunicationControl: return HandleCommunicationControl(req);
339 case uds::kReadMemoryByAddress: return HandleReadMemory(req);
340 case uds::kWriteMemoryByAddress: return HandleWriteMemory(req);
341 case uds::kRequestDownload:
342 case uds::kRequestUpload: return HandleRequestTransfer(req);
343 case uds::kTransferData: return HandleTransferData(req);
344 case uds::kRequestTransferExit: return HandleTransferExit(req);
345 default: return uds::NegativeResponse(sid, Nrc::kServiceNotSupported);
346 }
347 } catch (...) {
348 return uds::NegativeResponse(req[0], Nrc::kGeneralReject);
349 }
350}

380std::optional<ByteVector> DiagnosticServer::HandleRead(ByteVector const &req) {
381 constexpr auto sid = uds::kReadDataByIdentifier;
382 if (req.size() < 3 || (req.size() - 1) % 2 != 0)
383 return uds::NegativeResponse(sid, Nrc::kIncorrectMessageLengthOrInvalidFormat);
384 ByteVector out;
385 bool any = false;
386 for (std::size_t i = 1; i < req.size(); i += 2) {
387 std::uint16_t const did = Be16(req, i);
388 std::function<ara::core::Result<ByteVector, Nrc>()> read;
389 {
390 std::lock_guard<std::mutex> lk(mutex_);
391 auto it = dids_.find(did);
392 if (it == dids_.end() || !InSession(it->second.readSessions)) continue; // ISO: answer the DIDs that exist
393 read = it->second.read;
394 }
395 auto r = read();
396 if (!r.HasValue()) return uds::NegativeResponse(sid, r.Error());
397 PutBe16(out, did);
398 auto const &v = r.Value();
399 out.insert(out.end(), v.begin(), v.end());
400 any = true;
401 }
402 if (!any) return uds::NegativeResponse(sid, Nrc::kRequestOutOfRange); // none of the DIDs is available
403 return Positive(sid, std::move(out));
404}

408std::optional<ByteVector> DiagnosticServer::HandleCommunicationControl(ByteVector const &req) {
409 constexpr auto sid = uds::kCommunicationControl;
410 if (req.size() < 3) return uds::NegativeResponse(sid, Nrc::kIncorrectMessageLengthOrInvalidFormat);
411 std::uint8_t const sub = req[1] & 0x7F;
412 bool const suppress = (req[1] & uds::kSuppressPositiveResponseBit) != 0;
413 if (sub > 0x05) return uds::NegativeResponse(sid, Nrc::kSubFunctionNotSupported);
414 // sub-functions 0x04/0x05 carry a 2-byte nodeIdentificationNumber after the communicationType
415 if (req.size() != ((sub >= 0x04) ? 5U : 3U))
416 return uds::NegativeResponse(sid, Nrc::kIncorrectMessageLengthOrInvalidFormat);
417 std::uint8_t const type = req[2];
418 if ((type & 0xFC) != 0 || (type & 0x03) == 0) return uds::NegativeResponse(sid, Nrc::kRequestOutOfRange);
419 CommControlHandler h;
420 {
421 std::lock_guard<std::mutex> lk(mutex_);
422 h = commControl_;
423 }
424 if (!h) return uds::NegativeResponse(sid, Nrc::kServiceNotSupported);
425 if (!h(sub, type)) return uds::NegativeResponse(sid, Nrc::kConditionsNotCorrect);
426 {
427 std::lock_guard<std::mutex> lk(mutex_);
428 commState_.communicationType = type;
429 switch (sub) {
430 case 0x00: case 0x05: commState_.rxEnabled = true; commState_.txEnabled = true; break;
431 case 0x01: case 0x04: commState_.rxEnabled = true; commState_.txEnabled = false; break;
432 case 0x02: commState_.rxEnabled = false; commState_.txEnabled = true; break;
433 default: commState_.rxEnabled = false; commState_.txEnabled = false; break;
434 }
435 }
436 if (suppress) return std::nullopt;
437 return Positive(sid, {sub});
438}

451std::optional<ByteVector> DiagnosticServer::HandleReadMemory(ByteVector const &req) {
452 constexpr auto sid = uds::kReadMemoryByAddress;
453 std::size_t pos = 1;
454 std::uint64_t address = 0, length = 0;
455 if (!ParseAddressAndLength(req, pos, address, length) || pos != req.size())
456 return uds::NegativeResponse(sid, Nrc::kIncorrectMessageLengthOrInvalidFormat);
457 decltype(MemoryRegion::read) read;
458 {
459 std::lock_guard<std::mutex> lk(mutex_);
460 auto const *r = FindRegion(regions_, address, length);
461 if (r == nullptr) return uds::NegativeResponse(sid, Nrc::kRequestOutOfRange);
462 if (!InSession(r->sessions)) return uds::NegativeResponse(sid, Nrc::kServiceNotSupportedInActiveSession);
463 if (securityLevel_ < r->securityLevel) return uds::NegativeResponse(sid, Nrc::kSecurityAccessDenied);
464 read = r->read;
465 }
466 auto data = read(address, length);
467 if (!data.HasValue()) return uds::NegativeResponse(sid, data.Error());
468 if (data.Value().size() != length) return uds::NegativeResponse(sid, Nrc::kGeneralReject);
469 return Positive(sid, std::move(data).Value());
470}

473 constexpr auto sid = uds::kWriteMemoryByAddress;
474 std::size_t pos = 1;
475 std::uint64_t address = 0, length = 0;
476 if (!ParseAddressAndLength(req, pos, address, length) || req.size() - pos != length)
477 return uds::NegativeResponse(sid, Nrc::kIncorrectMessageLengthOrInvalidFormat);
478 ByteVector const data(req.begin() + static_cast<std::ptrdiff_t>(pos), req.end());

496std::optional<ByteVector> DiagnosticServer::HandleRequestTransfer(ByteVector const &req) {
497 std::uint8_t const sid = req[0];
498 std::size_t pos = 2;
499 std::uint64_t address = 0, size = 0;
500 if (req.size() < 3 || !ParseAddressAndLength(req, pos, address, size) || pos != req.size())
501 return uds::NegativeResponse(sid, Nrc::kIncorrectMessageLengthOrInvalidFormat);
502 TransferDirection const dir = sid == uds::kRequestDownload ? TransferDirection::kDownload : TransferDirection::kUpload;
503 TransferHandler h;
504 {
505 std::lock_guard<std::mutex> lk(mutex_);
506 if (!transfer_.request) return uds::NegativeResponse(sid, Nrc::kServiceNotSupported);
507 if (!InSession(transfer_.sessions)) return uds::NegativeResponse(sid, Nrc::kServiceNotSupportedInActiveSession);
508 if (securityLevel_ < transfer_.securityLevel) return uds::NegativeResponse(sid, Nrc::kSecurityAccessDenied);
509 if (xfer_.active) return uds::NegativeResponse(sid, Nrc::kConditionsNotCorrect);
510 h = transfer_;
511 }
512 if (size == 0) return uds::NegativeResponse(sid, Nrc::kRequestOutOfRange);
513 auto max = h.request(dir, address, size, req[1]);
514 if (!max.HasValue()) return uds::NegativeResponse(sid, max.Error());
515 if (max.Value() < 3) return uds::NegativeResponse(sid, Nrc::kGeneralReject);
516 std::uint32_t const block = std::min<std::uint32_t>(max.Value(), 0xFFFF);
517 {
518 std::lock_guard<std::mutex> lk(mutex_);
519 xfer_ = TransferState{true, dir, size, 0, block, 1};
520 }
521 ByteVector payload{0x20}; // lengthFormatIdentifier: maxNumberOfBlockLength is 2 bytes
522 PutBe16(payload, static_cast<std::uint16_t>(block));
523 return Positive(sid, std::move(payload));
524}

526std::optional<ByteVector> DiagnosticServer::HandleTransferData(ByteVector const &req) {
527 constexpr auto sid = uds::kTransferData;
528 if (req.size() < 2) return uds::NegativeResponse(sid, Nrc::kIncorrectMessageLengthOrInvalidFormat);
529 std::uint8_t const seq = req[1];
530 ByteVector const data(req.begin() + 2, req.end());
531 decltype(TransferHandler::block) block;
532 TransferState st;
533 {
534 std::lock_guard<std::mutex> lk(mutex_);
535 if (!xfer_.active) return uds::NegativeResponse(sid, Nrc::kRequestSequenceError);
536 if (seq != xfer_.nextCounter) return uds::NegativeResponse(sid, Nrc::kWrongBlockSequenceCounter);
537 st = xfer_;
538 block = transfer_.block;
539 }
540 if (!block) return uds::NegativeResponse(sid, Nrc::kServiceNotSupported);
541 if (st.dir == TransferDirection::kDownload) {
542 if (data.empty() || req.size() > st.maxBlock || st.done + data.size() > st.size)
543 return uds::NegativeResponse(sid, Nrc::kRequestOutOfRange);
544 } else if (!data.empty()) {
545 return uds::NegativeResponse(sid, Nrc::kIncorrectMessageLengthOrInvalidFormat);
546 }
547 auto r = block(seq, data);
548 if (!r.HasValue()) return uds::NegativeResponse(sid, r.Error());
549 std::uint64_t moved = data.size();
550 if (st.dir == TransferDirection::kUpload) {
551 moved = r.Value().size();
552 if (moved == 0 || moved + 2 > st.maxBlock || st.done + moved > st.size)
553 return uds::NegativeResponse(sid, Nrc::kGeneralReject);
554 }
555 {
556 std::lock_guard<std::mutex> lk(mutex_);
557 xfer_.done += moved;
558 xfer_.nextCounter = static_cast<std::uint8_t>(xfer_.nextCounter + 1); // wraps 0xFF -> 0x00
559 }
560 ByteVector payload{seq};
561 payload.insert(payload.end(), r.Value().begin(), r.Value().end());
562 return Positive(sid, std::move(payload));
563}

565std::optional<ByteVector> DiagnosticServer::HandleTransferExit(ByteVector const &req) {
566 constexpr auto sid = uds::kRequestTransferExit;
567 decltype(TransferHandler::exit) exit;
568 {
569 std::lock_guard<std::mutex> lk(mutex_);
570 if (!xfer_.active) return uds::NegativeResponse(sid, Nrc::kRequestSequenceError);
571 if (xfer_.dir == TransferDirection::kDownload && xfer_.done != xfer_.size)
572 return uds::NegativeResponse(sid, Nrc::kRequestSequenceError); // not all announced bytes were sent
573 exit = transfer_.exit;
574 }
575 ByteVector record;
576 if (exit) {
577 auto r = exit(ByteVector(req.begin() + 1, req.end()));
578 if (!r.HasValue()) return uds::NegativeResponse(sid, r.Error());
579 record = std::move(r).Value();
580 }
581 {
582 std::lock_guard<std::mutex> lk(mutex_);
583 xfer_ = TransferState{};
584 }
585 return Positive(sid, std::move(record));
586}
