40bool NmPduLayout::IsValid() const noexcept {
41 if (pduLength < 2 || pduLength > 64) return false;
42 if (nidPosition != kAbsent && (nidPosition < 0 || nidPosition > 1)) return false;
43 if (cbvPosition != kAbsent && (cbvPosition < 0 || cbvPosition > 1)) return false;
44 if (nidPosition != kAbsent && nidPosition == cbvPosition) return false;
45 if (pnVectorLength > 4) return false;
46 if (pnVectorLength > 0) {
47 if (cbvPosition == kAbsent) return false; // the PNI bit lives in the CBV
48 if (pnVectorOffset + pnVectorLength > pduLength) return false;
49 auto const lo = pnVectorOffset, hi = pnVectorOffset + pnVectorLength;
50 for (int fixed : {nidPosition, cbvPosition}) {
51 if (fixed != kAbsent && static_cast<std::size_t>(fixed) >= lo && static_cast<std::size_t>(fixed) < hi)
52 return false;
53 }
54 }
55 return true;

101ara::core::Result<NmPdu> NmPdu::Decode(std::uint8_t const *bytes, std::size_t size, NmPduLayout const &l) {
102 using R = ara::core::Result<NmPdu>;
103 if (!l.IsValid()) return R::FromError(MakeErrorCode(NmErrc::kInvalidConfig));
104 if (bytes == nullptr || size != l.pduLength) return R::FromError(MakeErrorCode(NmErrc::kBadPdu));
105 NmPdu p;
106 if (l.nidPosition != NmPduLayout::kAbsent) p.sourceNodeId = bytes[l.nidPosition];
107 if (l.cbvPosition != NmPduLayout::kAbsent) {
108 std::uint8_t const cbv = bytes[l.cbvPosition];
109 p.repeatMessageRequest = (cbv & kRmrBit) != 0;
110 p.partialNetworkInfo = (cbv & kPniBit) != 0 && l.pnVectorLength > 0;
111 }
112 for (std::size_t i = 0; i < l.pnVectorLength; ++i)
113 p.pnRequest |= static_cast<std::uint32_t>(bytes[l.pnVectorOffset + i]) << (8 * i);

116 for (std::size_t k = 0; k < idx.size(); ++k) {
117 if (k < 2) p.userData[k] = bytes[idx[k]];
118 else p.moreUserData.push_back(bytes[idx[k]]);

120 return R::FromValue(std::move(p));
