36std::string ToHex(ByteVector const &b) {
37 static constexpr char kDigits[] = "0123456789abcdef";
38 std::string s;
39 s.reserve(b.size() * 2);
40 for (auto v : b) {
41 s.push_back(kDigits[v >> 4]);
42 s.push_back(kDigits[v & 0xF]);
43 }
44 return s;
45}

47bool FromHex(std::string const &s, ByteVector &out) {
48 if (s.size() % 2 != 0) return false;
49 auto nib = [](char c) -> int {
50 if (c >= '0' && c <= '9') return c - '0';
51 if (c >= 'a' && c <= 'f') return c - 'a' + 10;
52 return -1;
53 };
54 out.clear();
55 for (std::size_t i = 0; i < s.size(); i += 2) {
56 int const hi = nib(s[i]);
57 int const lo = nib(s[i + 1]);
58 if (hi < 0 || lo < 0) return false;
59 out.push_back(static_cast<std::uint8_t>((hi << 4) | lo));
60 }
61 return true;
62}

108 // Builds the in-memory key from plaintext material and installs it.
109 static Res<void> Install(Slot &s, ByteView material) {
110 if (IsSymmetric(s.proto.algorithm)) {
111 if (s.proto.algorithm == SlotAlgorithm::kAes256Gcm && material.size() != 32)
112 return Res<void>::FromError(MakeErrorCode(CryptoErrc::kInvalidKeyLength));
113 if (s.proto.algorithm == SlotAlgorithm::kHmacSha256 && material.size() < 16)
114 return Res<void>::FromError(MakeErrorCode(CryptoErrc::kInvalidKeyLength));
115 auto k = SecretKey::Import(material, s.proto.usage & 0x7U, true);
116 if (!k.HasValue()) return Res<void>::FromError(k.Error());
117 s.key = std::move(k).Value();
118 } else {
119 auto k = PrivateKey::FromPkcs8Der(material);
120 if (!k.HasValue()) return Res<void>::FromError(k.Error());
121 if (k.Value().Algorithm() != SigAlg(s.proto.algorithm))
122 return Res<void>::FromError(MakeErrorCode(CryptoErrc::kIncompatibleArguments));
123 s.key = std::move(k).Value();
124 }
125 s.state = SlotState::kOccupied;
126 return {};
127 }

129 void Load(Slot &s) {
130 auto ex = storage->KeyExists(StorageKey(s.proto.name));
131 if (!ex.HasValue() || !ex.Value()) return;
132 auto hex = storage->GetValue<std::string>(StorageKey(s.proto.name));
133 ByteVector blob;
134 if (hex.HasValue() && FromHex(hex.Value(), blob)) {
135 auto plain = AeadOpen(wrap, Aad(s.proto), blob);
136 if (plain.HasValue()) {
137 auto r = Install(s, plain.Value());
138 SecureZero(plain.Value().data(), plain.Value().size());
139 if (r.HasValue()) return;
140 }
141 }
142 s.state = SlotState::kCorrupted;
143 }

145 // Finds the slot and checks it holds a usable key with the requested usage.
146 Res<Slot *> Use(std::string const &name, std::uint32_t usage) {
147 auto it = slots.find(name);
148 if (it == slots.end()) return Fail<Slot *>(CryptoErrc::kUnknownIdentifier);
149 Slot &s = it->second;
150 if (s.state == SlotState::kCorrupted) return Fail<Slot *>(CryptoErrc::kAuthTagNotValid);
151 if (s.state == SlotState::kEmpty) return Fail<Slot *>(CryptoErrc::kEmptyContainer);
152 if ((s.proto.usage & usage) != usage) return Fail<Slot *>(CryptoErrc::kUsageViolation);
153 return &s;
154 }

178Res<void> KeyStorageProvider::DefineSlot(KeySlotPrototype const &proto) {
179 auto bad = [](CryptoErrc e) { return Res<void>::FromError(MakeErrorCode(e)); };
180 if (!ValidName(proto.name)) return bad(CryptoErrc::kInvalidArgument);
181 if (proto.usage == 0 || (proto.usage & ~AllowedUsage(proto.algorithm)) != 0)
182 return bad(CryptoErrc::kIncompatibleArguments);
183 // Exporting a private signing key is not supported by this provider.
184 if (proto.exportable && !IsSymmetric(proto.algorithm)) return bad(CryptoErrc::kUnsupported);
185 std::lock_guard<std::mutex> lk(impl_->mutex);
186 if (proto.persistent && !impl_->storage) return bad(CryptoErrc::kUnsupported);
187 if (impl_->slots.count(proto.name) != 0) return bad(CryptoErrc::kInvalidArgument);
188 Slot &s = impl_->slots[proto.name];
189 s.proto = proto;
190 if (proto.persistent) impl_->Load(s);
191 return {};

208Res<void> KeyStorageProvider::Generate(std::string const &name) {
209 std::lock_guard<std::mutex> lk(impl_->mutex);
210 auto it = impl_->slots.find(name);
211 if (it == impl_->slots.end()) return Res<void>::FromError(MakeErrorCode(CryptoErrc::kUnknownIdentifier));
212 Slot &s = it->second;
213 ByteVector plain;
214 if (IsSymmetric(s.proto.algorithm)) {
