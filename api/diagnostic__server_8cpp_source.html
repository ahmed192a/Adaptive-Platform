37auto RetryPending(std::uint8_t sid, DiagnosticServer::ResponseSink const &sink, unsigned maxRetries, F &&call) {
38 auto r = call();
39 for (unsigned n = 0; !r.HasValue() && r.Error() == Nrc::kResponsePending && sink && n < maxRetries; ++n) {
40 sink(uds::NegativeResponse(sid, Nrc::kResponsePending));
41 r = call();
42 }
43 return r;

189std::optional<ByteVector> DiagnosticServer::HandleSecurityAccess(ByteVector const &req) {
190 constexpr auto sid = uds::kSecurityAccess;
191 if (req.size() < 2) return uds::NegativeResponse(sid, Nrc::kIncorrectMessageLengthOrInvalidFormat);
192 std::uint8_t const sub = req[1] & 0x7F;
193 bool const suppress = (req[1] & uds::kSuppressPositiveResponseBit) != 0;
194 if (sub == 0 || sub > 0x7E) return uds::NegativeResponse(sid, Nrc::kSubFunctionNotSupported);
195 SecurityProvider prov;
196 {
197 std::lock_guard<std::mutex> lk(mutex_);
198 prov = security_;
199 }
200 if (!prov.seed || !prov.validateKey) return uds::NegativeResponse(sid, Nrc::kServiceNotSupported);
201 auto const now = cfg_.clock();
202 if (sub % 2 == 1) { // requestSeed
203 std::uint8_t const level = sub;
204 {
205 std::lock_guard<std::mutex> lk(mutex_);
206 if (failedAttempts_ >= prov.maxAttempts && now < lockedUntil_)
207 return uds::NegativeResponse(sid, Nrc::kRequiredTimeDelayNotExpired);
208 if (securityLevel_ == level) { // already unlocked: all-zero seed of the usual length
209 ByteVector zeros(std::max<std::size_t>(1, pendingSeed_.size()), 0);
210 return Positive(sid, [&] { ByteVector p{sub}; p.insert(p.end(), zeros.begin(), zeros.end()); return p; }());
211 }
212 }
213 auto seed = prov.seed(level);
214 if (!seed.HasValue()) return uds::NegativeResponse(sid, seed.Error());
215 if (seed.Value().empty()) return uds::NegativeResponse(sid, Nrc::kGeneralReject);
216 ByteVector payload{sub};
217 payload.insert(payload.end(), seed.Value().begin(), seed.Value().end());
218 {
219 std::lock_guard<std::mutex> lk(mutex_);
220 pendingLevel_ = level;
221 pendingSeed_ = seed.Value();
222 }
223 return Positive(sid, std::move(payload)); // a seed response is never suppressed
224 }
225 // sendKey
226 if (req.size() < 3) return uds::NegativeResponse(sid, Nrc::kIncorrectMessageLengthOrInvalidFormat);
227 std::uint8_t const level = static_cast<std::uint8_t>(sub - 1);
228 ByteVector seed;
229 {
230 std::lock_guard<std::mutex> lk(mutex_);
231 if (pendingLevel_ != level) return uds::NegativeResponse(sid, Nrc::kRequestSequenceError);
232 if (failedAttempts_ >= prov.maxAttempts) {
233 if (now < lockedUntil_) return uds::NegativeResponse(sid, Nrc::kRequiredTimeDelayNotExpired);
234 failedAttempts_ = 0;
235 }
236 seed = pendingSeed_;
237 }
238 bool const ok = prov.validateKey(level, seed, ByteVector(req.begin() + 2, req.end()));
239 std::lock_guard<std::mutex> lk(mutex_);
240 pendingLevel_ = 0;
241 if (!ok) {
242 if (++failedAttempts_ >= prov.maxAttempts) {
243 lockedUntil_ = now + prov.lockout;
244 return uds::NegativeResponse(sid, Nrc::kExceededNumberOfAttempts);
245 }
246 return uds::NegativeResponse(sid, Nrc::kInvalidKey);
247 }
248 failedAttempts_ = 0;
249 securityLevel_ = level;
250 if (suppress) return std::nullopt;
251 return Positive(sid, {sub});
252}

254std::optional<ByteVector> DiagnosticServer::HandleRoutineControl(ByteVector const &req) {
255 constexpr auto sid = uds::kRoutineControl;
256 if (req.size() < 4) return uds::NegativeResponse(sid, Nrc::kIncorrectMessageLengthOrInvalidFormat);
