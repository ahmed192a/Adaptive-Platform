104std::vector<uint8_t> Entry::BasePayload(uint8_t &optionIndex) const {
105 std::vector<uint8_t> _result;
106
107 auto _type = static_cast<uint8_t>(Type());
108 _result.push_back(_type);
109
110 // Each run holds at most 15 options (4 bit counters).
111 if (FirstOptions().size() > 15 || SecondOptions().size() > 15) {
112 throw std::length_error("an SD entry can reference at most 15 options per run");
113 }
114 auto _firstOptionsSize = (uint8_t)FirstOptions().size();
115 auto _secondOptionsSize = (uint8_t)SecondOptions().size();
116
117 // An empty run must carry index 0 (option referencing rules,
118 // PRS_SOMEIPSD_00341-00343, 00833, 00834).
119 _result.push_back(_firstOptionsSize == 0 ? 0 : optionIndex);
120 optionIndex += _firstOptionsSize;
121
122 _result.push_back(_secondOptionsSize == 0 ? 0 : optionIndex);
123 optionIndex += _secondOptionsSize;
124
125 const uint8_t OptionSizeBitLength = 4;
126 _firstOptionsSize <<= OptionSizeBitLength;
127 _firstOptionsSize |= _secondOptionsSize;
128 _result.push_back(_firstOptionsSize);
129
130 helper::Inject(_result, ServiceId());
131 helper::Inject(_result, InstanceId());
132
133 const uint8_t TTLSizeBitLength = 24;
134 uint32_t _majorVersion = MajorVersion();
135 _majorVersion <<= TTLSizeBitLength;
136 _majorVersion |= TTL();
137 helper::Inject(_result, _majorVersion);
138
139 return _result;
140}
