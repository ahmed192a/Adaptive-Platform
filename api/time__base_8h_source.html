123constexpr SynchronizationStatus ToSynchronizationStatus(StatusFlags f) noexcept {
124 if (f.Test(StatusFlag::kTimeout)) return SynchronizationStatus::kTimeOut;
125 if (f.Test(StatusFlag::kSynchronized)) {
126 return f.Test(StatusFlag::kSyncToGateway) ? SynchronizationStatus::kSynchToGateway
127 : SynchronizationStatus::kSynchronized;
128 }
129 return SynchronizationStatus::kNotSynchronizedUntilStartup;
