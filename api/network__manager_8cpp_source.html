80Result<void> NetworkManager::InstallSigtermHook() {
81 std::lock_guard<std::mutex> l(gHookMu);
82 if (gInstalled) return Result<void>::FromValue();
83 struct sigaction sa {};
84 sa.sa_handler = NmSigtermHandler;
85 sigemptyset(&sa.sa_mask);
