52bool Message::IncrementSessionId() noexcept {
53 // Session ID range is [0x0001, 0xFFFF] and wraps to 0x0001
54 // [PRS_SOMEIP_00933, PRS_SOMEIP_00521].

58 if (header.session_id == SessionIdMax) {

122Message::TryDeserialize(const std::vector<uint8_t> &msg) noexcept {
123 if (msg.size() < HeaderSize) {
124 return std::nullopt;
125 }
126 header_t h;
127 h.service_id = Be16(msg, 0);
128 h.method_id = Be16(msg, 2);
129 h.length = Be32(msg, 4);
130 h.client_id = Be16(msg, 8);
131 h.session_id = Be16(msg, 10);
132 h.protocol_version = msg[12];
133 h.interface_version = msg[13];
134 h.message_type = msg[14];
135 h.return_code = msg[15];
136 // Length counts everything after the Length field [PRS_SOMEIP_00042].
137 if (h.length < EmptyLength ||
138 static_cast<uint64_t>(h.length) + 8 != msg.size()) {
139 return std::nullopt;
140 }
141 try {
142 return Message(h, std::vector<uint8_t>(msg.begin() + HeaderSize, msg.end()));
143 } catch (...) {
144 return std::nullopt;
145 }
146}
