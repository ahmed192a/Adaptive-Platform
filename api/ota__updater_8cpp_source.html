52// FNV-1a, 64 bit: stands in for the signature/digest a real package manifest carries.
53std::uint64_t Digest(std::vector<std::uint8_t> const &data) {
54 std::uint64_t h = 1469598103934665603ULL;
55 for (auto b : data) {
56 h ^= b;
57 h *= 1099511628211ULL;
58 }
59 return h;
60}

62std::vector<std::uint8_t> GeneratePackage(std::uint64_t size) {
63 std::vector<std::uint8_t> p(static_cast<std::size_t>(size));
64 std::uint32_t x = 0x12345678u;
65 for (auto &b : p) {
66 x = x * 1664525u + 1013904223u;
67 b = static_cast<std::uint8_t>(x >> 24);
68 }

103UpdateReport RunUpdate(UpdateOptions const &opt, std::ostream &out) {
104 UpdateReport rep;
105 std::ostringstream sink;
106 std::ostream &log = opt.quiet ? sink : out;
107
108 // ----------------------------------------------------------------- package
109 std::vector<std::uint8_t> package;
110 if (!opt.packageFile.empty()) {
111 std::ifstream f(opt.packageFile, std::ios::binary);
112 if (!f) {
113 rep.message = "cannot read package file " + opt.packageFile;
114 return rep;
115 }
116 package.assign(std::istreambuf_iterator<char>(f), std::istreambuf_iterator<char>());
117 } else {
118 package = GeneratePackage(opt.packageSize);
119 }
120 if (package.empty()) {
121 rep.message = "empty package";
122 return rep;
123 }
124 std::uint64_t const expectedDigest = Digest(package); // what the manifest promises
125
126 // ------------------------------------------------- State Management (the other side)
127 sm::StateManager manager;
128 sm::StateMachine &machine = manager.Add(
129 "Machine", {"Initial", "Off", "Driving", "PrepareUpdate", "VerifyUpdate", "PrepareRollback"}, "Initial");
130 machine.AddTransition("Initial", "Driving");
131 machine.AddTransition("Driving", "PrepareUpdate");
132 machine.AddTransition("PrepareUpdate", "VerifyUpdate");
133 machine.AddTransition("VerifyUpdate", "Driving");
134 machine.AddTransition("VerifyUpdate", "PrepareRollback");
135 machine.AddTransition("PrepareUpdate", "PrepareRollback");
136 machine.AddTransition("PrepareRollback", "Driving");
137 machine.AddTransition("*", "Off");
138 machine.SetNotifier([&](std::string const &from, std::string const &to) {
139 rep.machineTrail.push_back(to);
140 log << " [SM] machine state " << from << " -> " << to << "\n";
141 });
142 rep.machineTrail.push_back("Initial");
143 manager.Start("Machine", "Driving");
144 ara::sm::update_request::UpdateSession session;
145
146 // ------------------------------------------------------------------ UCM core
147 core::PackageManagerCore ucm;
148 ucm.SetId("ucm-demo");
149 rep.statusTrail.push_back(StatusName(ucm.Status()));
150 ucm.OnStatusChange([&](PM::PackageManagerStatusType s) {
151 rep.statusTrail.push_back(StatusName(s));
152 log << " [UCM] CurrentStatus = " << StatusName(s) << "\n";
153 });
154
155 std::map<PM::TransferIdType, std::vector<std::uint8_t>> received; // what the UCM stored on "flash"
156 core::UpdateHooks hooks;
157 hooks.validatePackage = [&](PM::TransferIdType const &id) {
158 auto it = received.find(id);
159 return it != received.end() && Digest(it->second) == expectedDigest ? 0 : static_cast<int>(PM::PackageInconsistent);
160 };
161 hooks.prepareUpdate = [&]() {
162 if (opt.rejectSession) return static_cast<int>(PM::UpdateSessionRejected);
163 if (session.RequestUpdateSession() || session.PrepareUpdate()) return static_cast<int>(PM::UpdateSessionRejected);
164 return machine.RequestTransition("PrepareUpdate") ? static_cast<int>(PM::PreActivationFailed) : 0;
165 };
166 hooks.resetMachine = [&]() {
167 if (session.ResetMachine()) return static_cast<int>(PM::PreActivationFailed);
168 log << " [SM] ResetMachine requested; UCM state is persisted across the restart\n";
169 if (!opt.stateFile.empty() && !ucm.Save(opt.stateFile)) return static_cast<int>(PM::PersistencyAllocationFailed);
170 return 0;
171 };
172 hooks.verifyUpdate = [&]() {
173 if (session.VerifyUpdate() || machine.RequestTransition("VerifyUpdate")) return 1;
174 return opt.failVerify ? 1 : 0;
175 };
176 hooks.prepareRollback = [&]() {
177 if (session.PrepareRollback() || machine.RequestTransition("PrepareRollback")) return 1;
178 return 0;
179 };
180 ucm.SetHooks(hooks);
181
182 auto fail = [&](Outcome o, int err, std::string msg) {
183 rep.outcome = o;
184 rep.ucmError = err;
185 rep.message = std::move(msg);
186 rep.ucmStatus = StatusName(ucm.Status());
187 rep.machineState = machine.CurrentState();
188 log << " result: " << rep.message << "\n";
189 return rep;
190 };
191
192 // ----------------------------------------------------------------- transfer
193 PM::TransferIdType id{};
194 for (std::size_t i = 0; i < id.size(); ++i) id[i] = static_cast<std::uint8_t>(0xA0 + i);
195 log << "1. Transfer " << package.size() << " bytes\n";
196 auto start = ucm.TransferStart(package.size(), id);
197 if (start.error != 0) return fail(Outcome::kError, start.error, std::string("TransferStart: ") + ErrorName(start.error));
198 auto &stored = received[id];
199 std::uint64_t sent = 0;
200 std::uint64_t block = 0;
201 std::uint64_t const total = package.size();
202 while (sent < total) {
203 std::uint64_t const n = std::min<std::uint64_t>(start.blockSize, total - sent);
204 std::vector<std::uint8_t> chunk(package.begin() + static_cast<std::ptrdiff_t>(sent),
205 package.begin() + static_cast<std::ptrdiff_t>(sent + n));
206 if (opt.tamper && block == 0) chunk[0] = static_cast<std::uint8_t>(chunk[0] ^ 0xFF); // corrupted in transit
207 auto r = ucm.TransferData(id, n, block);
208 if (r.error != 0) return fail(Outcome::kError, r.error, std::string("TransferData: ") + ErrorName(r.error));
209 stored.insert(stored.end(), chunk.begin(), chunk.begin() + static_cast<std::ptrdiff_t>(r.accepted));
210 sent += r.accepted;
211 ++block;
212 if (!opt.quiet && (block % 4 == 0 || sent == total)) Bar(log, "transfer", sent, total);
213 }
214 rep.blocks = block;
215 int e = ucm.TransferExit(id);
216 if (e != 0) {
217 ucm.DeleteTransfer(id);
218 return fail(Outcome::kPackageRejected, e,
219 std::string("TransferExit: ") + ErrorName(e) + " (digest mismatch) - transfer deleted");
220 }
221 log << "2. Package verified (digest 0x" << std::hex << expectedDigest << std::dec << ")\n";
222
223 // --------------------------------------------------------------- processing
224 log << "3. ProcessSwPackage\n";
225 e = ucm.BeginProcess(id);
226 if (e != 0) return fail(Outcome::kError, e, std::string("ProcessSwPackage: ") + ErrorName(e));
227 {
228 std::uint64_t n = 0;
229 for (char const *step : {"unpack", "check", "write flash"}) Bar(log, step, ++n, 3);
230 }
231 ucm.EndProcess(id, true);
232
233 // --------------------------------------------------------------- activation
234 log << "4. Activate (State Management: PrepareUpdate, ResetMachine)\n";
235 e = ucm.Activate();
236 if (e != 0) {
237 ucm.Revert();
238 session.StopUpdateSession();
239 return fail(Outcome::kSessionRejected, e, std::string("Activate: ") + ErrorName(e) + " - packages reverted");
240 }
241 log << "5. Verify the new software\n";
242 e = ucm.Verify();
243 if (e != 0) {
244 log << " verification failed -> Rollback\n";
245 int const rb = ucm.Rollback();
246 machine.RequestTransition("Driving");
247 session.StopUpdateSession();
248 ucm.BeginFinish();
249 ucm.EndFinish();
250 if (rb != 0) return fail(Outcome::kError, rb, std::string("Rollback: ") + ErrorName(rb));
251 return fail(Outcome::kRolledBack, e, "update rolled back: " + std::string(ErrorName(e)));
252 }
253 log << "6. Finish\n";
254 ucm.BeginFinish();
255 ucm.EndFinish();
256 machine.RequestTransition("Driving");
257 session.StopUpdateSession();
258
259 rep.outcome = Outcome::kUpdated;
260 rep.message = "update installed";
261 rep.ucmStatus = StatusName(ucm.Status());
262 rep.machineState = machine.CurrentState();
263 log << " result: " << rep.message << "\n";
264 return rep;
265}
