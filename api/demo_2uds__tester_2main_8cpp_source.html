2// uds_tester scripted session, prints a readable trace (exit code 1 when a check fails)
3// uds_tester --repl type hex requests ("22 F1 90"); "f " prefix = functional; "wait <ms>" advances the S3 clock

13 std::string a = argv[i];

24
25 std::cout << "demo ECU ready. Hex requests, e.g. '10 03', '22 F1 90'. 'f <hex>' = functional, 'wait <ms>', 'quit'.\n";
26 std::string line;
27 while (std::cout << "> " && std::getline(std::cin, line)) {
28 if (line == "quit" || line == "exit") break;
29 if (line.rfind("wait ", 0) == 0) {
30 ecu.Advance(std::chrono::milliseconds(std::stoi(line.substr(5))));
31 continue;
32 }
33 auto addr = ara::diag::Addressing::kPhysical;
34 if (line.rfind("f ", 0) == 0) {
35 addr = ara::diag::Addressing::kFunctional;
36 line = line.substr(2);
37 }
38 auto req = ap::demo::ParseHex(line);
39 if (!req) {
40 std::cout << " not a hex request\n";
41 continue;
42 }
43 auto rsp = ecu.Server().Dispatch(*req, addr);
44 std::cout << " " << ap::demo::Describe(*req) << "\n";
45 if (rsp) std::cout << " RX " << ap::demo::HexString(*rsp) << " " << ap::demo::Describe(*rsp) << "\n";
46 else std::cout << " (no response)\n";
47 }

49}
