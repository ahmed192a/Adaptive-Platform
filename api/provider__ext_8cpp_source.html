13Res<void> KeyDerivationFunctionCtx::Configure(ByteView salt, ByteView info, std::uint32_t iterations) {
14 if (kind_ == KdfKind::kPbkdf2 && (iterations == 0 || salt.empty())) return FailV(CryptoErrc::kInvalidArgument);
15 salt_.assign(salt.data(), salt.data() + salt.size());
16 info_.assign(info.data(), info.data() + info.size());
17 iterations_ = iterations;
18 configured_ = true;
19 return {};
20}
21Res<ByteVector> KeyDerivationFunctionCtx::DeriveBytes(ByteView secret, std::size_t length) const {
22 if (!configured_) return Fail<ByteVector>(CryptoErrc::kUninitializedContext);
23 return kind_ == KdfKind::kHkdf ? Hkdf(alg_, secret, salt_, info_, length)
24 : Pbkdf2(alg_, secret, salt_, iterations_, length, weak_);
25}
26Res<SecretKey> KeyDerivationFunctionCtx::DeriveKey(ByteView secret, std::size_t keySize, std::uint32_t usage,
27 bool exportable) const {
28 auto b = DeriveBytes(secret, keySize);
29 if (!b.HasValue()) return Res<SecretKey>::FromError(b.Error());
30 auto k = SecretKey::Import(b.Value(), usage, exportable);
31 SecureZero(b.Value().data(), b.Value().size());
32 return k;
33}
