21E::TransferIdType MakeId(std::uint8_t seed) {
22 E::TransferIdType id{};
23 for (std::size_t i = 0; i < id.size(); ++i) id[i] = static_cast<std::uint8_t>(seed + i);
24 return id;
25}
26
27// Uploads a 2500 byte package (3 blocks of 1024) and returns its id; the core ends in kTransferred.
28E::TransferIdType Upload(PackageManagerCore &c, std::uint8_t seed = 1) {
29 auto id = MakeId(seed);
30 auto s = c.TransferStart(2500, id);
31 EXPECT_EQ(s.error, 0);
32 EXPECT_EQ(c.TransferData(id, 1024, 0).error, 0);
33 EXPECT_EQ(c.TransferData(id, 1024, 1).error, 0);
34 EXPECT_EQ(c.TransferData(id, 452, 2).error, 0);
35 return id;
36}
37
38E::TransferIdType ProcessOne(PackageManagerCore &c, std::uint8_t seed = 1) {
39 auto id = Upload(c, seed);
40 EXPECT_EQ(c.TransferExit(id), 0);
41 EXPECT_EQ(c.BeginProcess(id), 0);
42 c.EndProcess(id, true);
43 return id;
44}
