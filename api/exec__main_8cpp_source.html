221void monitor_children()
222{
223 for (auto &p : process_pool) {
224 if (!p.prun || p._pid <= 0) continue;
225 int status = 0;
226 if (waitpid(p._pid, &status, WNOHANG) != p._pid) continue;
227 std::string fg = "MachineFG";
228 if (p.current_config && !p.current_config->machine_instance_refs.empty())
229 fg = p.current_config->machine_instance_refs.front().function_group;
230 bool const clean = !ara::exec::IsAbnormalTermination(status);
231 if (clean)
232 cout << "- EM: process " << p.name << " exited on its own (status 0)" << endl;
233 else
234 cout << "- EM: ERROR: process " << p.name << " terminated unexpectedly (status " << status
235 << ") in function group " << fg << endl;
236 close(p.exec_clinet_fd);
237 unlink(("processes/" + p.name + "/execution_client_fifo").c_str());
238 p.prun = false;
239 p.current_config = nullptr;
240 p._pid = 0;
241 if (!clean) publish_execution_error(fg, ara::exec::kDefaultExecutionError);
242 }

277void p_operator(){
278 for(auto &process : process_pool){
279 if(process.prun|| process._pid != 0)continue; // if the process is running continue
280 bool will_run = false; // flag to indicate if the process will run or not
281 //check if the current state doesnt violate the configuration,otherwise terminate
282 for(int config = 0; config < process.startup_configs.size();config++){
283 bool violate = false; // flag to indicate if the current state violates the configuration
284 // for each machine instance(FG) reference in the current configuration
285 for(auto &mref: process.startup_configs[config].machine_instance_refs){
286 //Get the current state of the FG
287 string fg_statue = sys_FG[mref.function_group].current_FGS->get_states();
288 //Process wont run if fg_state is not process_config_states
289 if (!count(mref.modes.begin(), mref.modes.end(), fg_statue)){
290 violate = true;
291 break;
292 }
293 }
294 if(!violate){ // if the current state doesnt violate the configuration
295 process.current_config = &process.startup_configs[config]; // set the current configuration
296 will_run = true; // set the flag to true to indicate that the process will run
297 break;
298 }
299 }
