95// Registers the single shared observer that fans out to the three notifier kinds.
96template <class ImplT> void EnsureObserver(std::shared_ptr<ImplT> const &impl, std::shared_ptr<TimeBaseResource> const &r) {
97 if (impl->observerId != 0) return;
98 std::weak_ptr<ImplT> weak = impl;
99 impl->observed = r;
100 impl->observerId = r->AddObserver(
101 [weak](TimeBaseSnapshot const &snap, ChangeReason reason) {
102 auto self = weak.lock();
103 if (!self) return;
104 std::function<void(SynchronizedTimeBaseStatus const &)> st;
105 std::function<void(SynchronizationStatus)> sy;
106 std::function<void(LeapJump)> lp;
107 bool statusChanged = false, syncChanged = false;
108 {
109 std::lock_guard<std::mutex> g(self->mu);
110 st = self->onStatus;
111 sy = self->onSync;
112 lp = self->onLeap;
113 statusChanged = snap.status != self->lastFlags;
114 syncChanged = ToSynchronizationStatus(snap.status) != ToSynchronizationStatus(self->lastFlags) ||
115 !self->lastEver;
116 self->lastFlags = snap.status;
117 self->lastEver = true;
118 }
119 SynchronizedTimeBaseStatus status(snap, true);
120 if (st && statusChanged) st(status);
121 if (sy && syncChanged) sy(status.GetSynchronizationStatus());
122 if (lp && reason == ChangeReason::kTimeStep) lp(snap.leapJump);
123 },
124 /*notifyEveryUpdate=*/true);
125}
