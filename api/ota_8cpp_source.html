270 unsigned int clntLen = sizeof(echoClntAddr); /* Address length */
271
272 std::vector<uint8_t> msg; /* Buffer for storing received datagram */

274
275 /* Receive size of the message */

278 msg.resize(msg_size); /* Size the buffer that is received into */
279 /* Receive the message */

282 cout << "\t[CLIENT] ->> Handling client "

288 cudp.addr =

290
