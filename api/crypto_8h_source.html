211template <typename Fn> auto KeyStore::With(std::string const &slot, Fn &&fn) const
212 -> Res<decltype(fn(std::declval<SecretKey const &>()))> {
213 using R = decltype(fn(std::declval<SecretKey const &>()));
214 std::lock_guard<std::mutex> lk(mutex_);
215 auto it = slots_.find(slot);

217 return Res<R>::FromValue(fn(it->second));
