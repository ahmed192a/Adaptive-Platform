76 static std::unique_ptr<Vault> Open(fs::path const &dir, std::string const &pass, std::string &err) {
77 std::error_code ec;
78 fs::create_directories(dir, ec);
79 ByteVector salt;
80 if (!ReadFile(dir / "salt.bin", salt)) {
81 auto s = GenerateRandom(16);
82 if (!s.HasValue() || !WriteFile(dir / "salt.bin", s.Value())) { err = "cannot create salt"; return nullptr; }
83 salt = s.Value();
84 }
85 auto wrap = Pbkdf2(HashAlgorithm::kSha256, ByteView(pass), salt, kPbkdf2Iterations, 32);
86 if (!wrap.HasValue()) { err = "kdf: " + Err(wrap); return nullptr; }
87 auto cfg = ara::per::PersistencyConfig::FromJson(R"({"storageRoot":")" + (dir / "per").string() +
88 R"(","storages":[{"specifier":"Vault/Keys","type":"kvs"}]})");
89 if (!cfg.HasValue() || !ara::per::Configure(std::move(cfg).Value()).HasValue()) { err = "persistency config"; return nullptr; }
90 auto kvs = ara::per::OpenKeyValueStorage(ara::core::InstanceSpecifier("Vault/Keys"));
91 if (!kvs.HasValue()) { err = "open storage"; return nullptr; }
92 auto prov = KeyStorageProvider::Create(kvs.Value(), wrap.Value());
93 SecureZero(wrap.Value().data(), wrap.Value().size());
94 if (!prov.HasValue()) { err = "key storage: " + Err(prov); return nullptr; }
95 auto v = std::unique_ptr<Vault>(new Vault(std::move(prov).Value()));
96 KeySlotPrototype aes{kAesSlot, SlotAlgorithm::kAes256Gcm, kSlotEncrypt | kSlotDecrypt, false, true};
97 KeySlotPrototype sig{kSignSlot, SlotAlgorithm::kEd25519, kSlotSign | kSlotVerify, false, true};
98 if (!v->keys.DefineSlot(aes).HasValue() || !v->keys.DefineSlot(sig).HasValue()) { err = "define slots"; return nullptr; }
99 return v;
100 }

107bool EnsureKeys(Vault &v, Trace &t) {
108 for (char const *name : {kAesSlot, kSignSlot}) {
109 auto info = v.keys.GetSlotInfo(name);
110 if (!info.HasValue()) return false;
111 if (info.Value().state == SlotState::kCorrupted) {
112 t.Check(false, std::string("slot ") + name + " is corrupted (wrong passphrase or tampered store)");
113 return false;
114 }
115 if (info.Value().state == SlotState::kEmpty) {
116 auto g = v.keys.Generate(name);
117 t.Check(g.HasValue(), std::string("generated key in slot ") + name);
118 if (!g.HasValue()) return false;
119 } else {
120 t.Info(std::string("slot ") + name + " already holds a key");
121 }
122 }
123 return true;
124}

127int RunPki(fs::path const &dir, Trace &t) {
128 t.Step("X.509 chain: root (Ed25519) -> intermediate (ECDSA P-256) -> leaf");
129 auto rootKey = PrivateKey::Generate(SignatureAlgorithm::kEd25519).Value();
130 auto caKey = PrivateKey::Generate(SignatureAlgorithm::kEcdsaP256Sha256).Value();
131 auto leafKey = PrivateKey::Generate(SignatureAlgorithm::kEcdsaP256Sha256).Value();
132 auto root = IssueCertificate({"Vault Root CA", rootKey.GetPublicKey().Value().ToDer().Value(), 1, 0, 0, true}, nullptr, rootKey);
133 auto ca = root.HasValue() ? IssueCertificate({"Vault Intermediate CA", caKey.GetPublicKey().Value().ToDer().Value(), 2, 0, 0, true},
134 &root.Value(), rootKey)
135 : root;
136 auto leaf = ca.HasValue() ? IssueCertificate({"ecu-1.vault", leafKey.GetPublicKey().Value().ToDer().Value(), 3, 0, 0, false},
137 &ca.Value(), caKey)
138 : ca;
139 t.Check(root.HasValue() && ca.HasValue() && leaf.HasValue(), "issued three certificates");
140 if (!leaf.HasValue()) return 1;
141 for (auto const *c : {&root.Value(), &ca.Value(), &leaf.Value()})
142 t.Info(c->SubjectName() + " <- " + c->IssuerName() + " serial=" + c->SerialNumberHex() + " CA=" + (c->IsCa() ? "yes" : "no") +
143 " sha256=" + Hex(c->Fingerprint().Value()));
144 if (auto pem = leaf.Value().ToPem(); pem.HasValue()) {
145 std::ofstream(dir / "leaf.pem") << pem.Value();
146 t.Info("wrote " + (dir / "leaf.pem").string());
147 }
148
149 X509Provider provider;
150 t.Check(provider.AddTrustAnchor(root.Value()).HasValue(), "root added as trust anchor");
151 t.Check(provider.VerifyChain(leaf.Value(), {ca.Value()}).HasValue(), "chain leaf -> intermediate -> root validates");
152 auto missing = provider.VerifyChain(leaf.Value());
153 t.Check(!missing.HasValue(), "chain without the intermediate is rejected (x509 error " +
154 std::to_string(missing.HasValue() ? 0 : missing.Error().SupportData()) + ")");
155 std::time_t const future = leaf.Value().NotAfter() + 86400;
156 t.Check(!provider.VerifyChain(leaf.Value(), {ca.Value()}, future).HasValue(), "an expired leaf is rejected");
157 auto der = leaf.Value().ToDer().Value();
158 der[der.size() - 4] ^= 0x01;
159 auto tampered = X509Certificate::FromDer(der);
160 t.Check(!tampered.HasValue() || !provider.VerifyChain(tampered.Value(), {ca.Value()}).HasValue(),
161 "a leaf with a flipped signature bit is rejected");
162 return 0;
163}

165int RunDemo(fs::path const &dir, Trace &t) {
166 std::error_code ec;
167 fs::remove_all(dir, ec);
168 fs::create_directories(dir, ec);
169 std::string const pass = "correct horse battery staple";
170 std::string err;
171
172 t.Step("1. create the vault: passphrase -> PBKDF2 -> wrapping key -> persistent key slots");
173 auto vault = Vault::Open(dir, pass, err);
174 t.Check(vault != nullptr, "vault opened " + err);
175 if (!vault) return 1;
176 if (!EnsureKeys(*vault, t)) return 1;
177 for (auto const &s : vault->keys.ListSlots())
178 t.Info("slot " + s.prototype.name + (s.prototype.persistent ? " (persistent)" : "") + (s.prototype.exportable ? " exportable" : " non-exportable"));
179
180 t.Step("2. seal a file with AES-256-GCM (key never leaves the slot)");
181 std::string const message = "Firmware 2.4.1 release notes: brake controller timing fix.";
182 auto sealed = vault->keys.Seal(kAesSlot, ByteView(std::string("doc-v1")), ByteView(message));
183 t.Check(sealed.HasValue(), "sealed " + std::to_string(message.size()) + " bytes -> " + std::to_string(sealed.HasValue() ? sealed.Value().size() : 0) + " bytes (nonce+ciphertext+tag)");
184 if (!sealed.HasValue()) return 1;
185 (void)WriteFile(dir / "notes.sealed", sealed.Value());
186 auto opened = vault->keys.Open(kAesSlot, ByteView(std::string("doc-v1")), sealed.Value());
187 t.Check(opened.HasValue() && std::string(opened.Value().begin(), opened.Value().end()) == message, "unsealed and compared with the original");
188 auto wrongAad = vault->keys.Open(kAesSlot, ByteView(std::string("doc-v2")), sealed.Value());
189 t.Check(!wrongAad.HasValue(), "wrong associated data is rejected");
190 auto flipped = sealed.Value();
191 flipped[20] ^= 0x80;
192 t.Check(!vault->keys.Open(kAesSlot, ByteView(std::string("doc-v1")), flipped).HasValue(), "a flipped ciphertext bit is rejected");
193
194 t.Step("3. sign and verify with Ed25519");
195 auto sig = vault->keys.Sign(kSignSlot, ByteView(message));
196 t.Check(sig.HasValue(), "signature " + Hex(sig.HasValue() ? sig.Value() : ByteVector{}, 12));
197 if (!sig.HasValue()) return 1;
198 t.Check(vault->keys.Verify(kSignSlot, ByteView(message), sig.Value()).ValueOr(false), "signature verifies");
199 t.Check(!vault->keys.Verify(kSignSlot, ByteView(message + "!"), sig.Value()).ValueOr(true), "signature over a modified message does not verify");
200
201 t.Step("4. restart: keys survive in the wrapped store");
202 auto pubBefore = vault->keys.GetPublicKeyDer(kSignSlot).Value();
203 vault.reset();
204 vault = Vault::Open(dir, pass, err);
205 t.Check(vault != nullptr, "vault reopened with the same passphrase");
206 if (!vault) return 1;
207 t.Check(vault->keys.GetPublicKeyDer(kSignSlot).ValueOr(ByteVector{}) == pubBefore, "same signing key after restart");
208 auto again = vault->keys.Open(kAesSlot, ByteView(std::string("doc-v1")), sealed.Value());
209 t.Check(again.HasValue(), "file sealed before the restart still opens");
210 vault.reset();
211 auto wrong = Vault::Open(dir, "wrong passphrase", err);
212 bool rejected = wrong == nullptr;
213 if (wrong) {
214 auto info = wrong->keys.GetSlotInfo(kAesSlot);
215 rejected = info.HasValue() && info.Value().state == SlotState::kCorrupted;
216 }
217 t.Check(rejected, "a wrong passphrase cannot read the slots");
218
219 t.Step("5. RSA-OAEP wraps a fresh AES key for another party");
220 auto rsa = RsaPrivateKey::Generate(2048).Value();
221 auto session = SecretKey::Generate(32, kUsageEncrypt | kUsageDecrypt, true).Value();
222 auto wrapped = rsa.GetPublicKey().Value().EncryptOaep(session.Export().Value());
223 t.Check(wrapped.HasValue(), "session key wrapped for the recipient (" + std::to_string(wrapped.HasValue() ? wrapped.Value().size() : 0) + " bytes)");
224 auto unwrapped = rsa.DecryptOaep(wrapped.Value());
225 t.Check(unwrapped.HasValue() && unwrapped.Value() == session.Export().Value(), "recipient recovers the same session key");
226
227 int rc = RunPki(dir, t);
228 if (rc != 0) return rc;
229
230 std::cout << "\n" << (t.failures == 0 ? "secure_vault demo: ALL CHECKS PASSED" : "secure_vault demo: " + std::to_string(t.failures) + " CHECK(S) FAILED")
231 << std::endl;
232 return t.failures == 0 ? 0 : 1;

258int main(int argc, char **argv) {
259 Args a = Parse(argc, argv);
260 if (a.cmd.empty() || a.dir.empty()) return Usage();
261 Trace t;
262 fs::path const dir = a.dir;
263 if (a.cmd == "demo") return RunDemo(dir, t);
264 if (a.cmd == "pki") { std::error_code ec; fs::create_directories(dir, ec); int rc = RunPki(dir, t); return rc != 0 || t.failures != 0 ? 1 : 0; }
265 if (a.pass.empty()) { std::cerr << "--pass is required\n"; return 2; }
266 t.quiet = true;
267 std::string err;
268 auto vault = Vault::Open(dir, a.pass, err);
269 if (!vault) { std::cerr << "error: " << err << '\n'; return 1; }
270 if (!EnsureKeys(*vault, t)) return 1;
271 if (a.cmd == "init") return 0;
272 if (a.pos.size() != 2) return Usage();
273 ByteVector in;
274 if (!ReadFile(a.pos[0], in)) { std::cerr << "cannot read " << a.pos[0] << '\n'; return 1; }
275 if (a.cmd == "seal" || a.cmd == "unseal") {
276 auto r = a.cmd == "seal" ? vault->keys.Seal(kAesSlot, ByteView(std::string("vault")), in) : vault->keys.Open(kAesSlot, ByteView(std::string("vault")), in);
277 if (!r.HasValue()) { std::cerr << a.cmd << " failed: " << Err(r) << '\n'; return 1; }
278 return WriteFile(a.pos[1], r.Value()) ? 0 : 1;
279 }
280 if (a.cmd == "sign") {
281 auto r = vault->keys.Sign(kSignSlot, in);
282 if (!r.HasValue()) { std::cerr << "sign failed: " << Err(r) << '\n'; return 1; }
283 return WriteFile(a.pos[1], r.Value()) ? 0 : 1;
284 }
285 if (a.cmd == "verify") {
286 ByteVector sig;
287 if (!ReadFile(a.pos[1], sig)) { std::cerr << "cannot read " << a.pos[1] << '\n'; return 1; }
288 auto r = vault->keys.Verify(kSignSlot, in, sig);
289 bool ok = r.HasValue() && r.Value();
290 std::cout << (ok ? "signature OK" : "signature INVALID") << std::endl;
291 return ok ? 0 : 1;
292 }
293 return Usage();
