108TEST(SmModel, NewValidRequestCancelsTheRunningOne) { // 00607 / 00606
109 StateMachine sm("M", {"A", "B", "C"}, "A");
110 std::promise<void> started, release;
111 auto releaseF = release.get_future().share();
112 sm.AddTransition("A", "B", {{"slow", [&] { started.set_value(); releaseF.wait(); return true; }},
113 {"after", [] { return true; }}});
114 sm.AddTransition("A", "C");
115 auto first = std::async(std::launch::async, [&] { return sm.RequestTransition("B"); });
116 started.get_future().wait();
117 EXPECT_FALSE(sm.RequestTransition("C").has_value()); // newer request wins
118 release.set_value();
119 EXPECT_EQ(first.get(), TransitionResult(TransitionErrc::kOperationCanceled));
120 EXPECT_EQ(sm.CurrentState(), "C");
