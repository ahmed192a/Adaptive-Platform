24std::int64_t TimeBaseResource::TimeAtLocked(State const &s, SteadyTp now) const {
25 std::int64_t const dt = Ns(std::chrono::duration_cast<Duration>(now - s.localAtSync));
26 auto const scaled = static_cast<std::int64_t>(std::llround(static_cast<long double>(dt) * (1.0L + s.rate)));
27 std::int64_t applied = s.pendingOffsetNs;
28 if (s.pendingOffsetNs != 0) {
29 std::int64_t const win = Ns(std::chrono::duration_cast<Duration>(s.slewEnd - s.localAtSync));
30 if (win > 0 && dt < win) {
31 applied = static_cast<std::int64_t>(
32 std::llround(static_cast<long double>(s.pendingOffsetNs) * dt / static_cast<long double>(win)));
33 }
34 }
35 return s.baseNs + scaled + applied;
36}

38StatusFlags TimeBaseResource::StatusLocked(State const &s, SteadyTp now) const {
39 if (!s.everSynced) return {};
40 std::uint32_t bits = 0;
41 bool const timedOut = (now - s.lastUpdate) > config_.timeout;
42 bits |= timedOut ? static_cast<std::uint32_t>(StatusFlag::kTimeout)
43 : static_cast<std::uint32_t>(StatusFlag::kSynchronized);
44 if (s.gateway) bits |= static_cast<std::uint32_t>(StatusFlag::kSyncToGateway);
45 if (s.leap) bits |= static_cast<std::uint32_t>(StatusFlag::kTimeLeap);
46 if (s.pendingOffsetNs != 0 && now < s.slewEnd) bits |= static_cast<std::uint32_t>(StatusFlag::kPendingRateCorr);
47 return StatusFlags(bits);
48}

50TimeBaseSnapshot TimeBaseResource::SnapshotLocked(SteadyTp now) const {
51 TimeBaseSnapshot snap;
52 snap.time = SynchronizedTimePoint(Duration(TimeAtLocked(state_, now)));
53 snap.rateDeviation = state_.rate;
54 snap.status = StatusLocked(state_, now);
55 snap.userData = state_.user;
56 snap.localTime = now;
57 snap.rateCorrected = state_.rateCorrected;
58 snap.rateExceeded = config_.rateCorrectionThreshold > 0.0 && std::fabs(state_.rate) > config_.rateCorrectionThreshold;
59 if (state_.leap) snap.leapJump = state_.leapDir >= 0 ? LeapJump::kTimeLeapFuture : LeapJump::kTimeLeapPast;

99ara::core::Result<void> TimeBaseResource::Apply(SynchronizedTimePoint t, double rate, bool gateway,
100 UserData user, bool forceStep) {
101 if (!std::isfinite(rate) || rate <= -1.0 || user.length > 3) {
102 return ara::core::Result<void>::FromError(MakeErrorCode(TsyncErrc::kInvalidArgument));
103 }
104 auto const now = Now();
105 TimeBaseSnapshot snap;
106 bool stepped = false;
107 {
108 std::lock_guard<std::mutex> lk(mutex_);
109 std::int64_t const target = t.time_since_epoch().count();
110 if (!state_.everSynced) {
111 state_.baseNs = target;
112 state_.pendingOffsetNs = 0;
113 state_.leap = false;
114 state_.leapDir = 0;
115 } else {
116 std::int64_t const current = TimeAtLocked(state_, now);
117 std::int64_t const offset = target - current;
118 bool const beyondThreshold = std::llabs(offset) > Ns(config_.stepThreshold);
119 if (beyondThreshold || forceStep) {
120 state_.baseNs = target;
121 state_.pendingOffsetNs = 0;
122 state_.leap = beyondThreshold;
123 state_.leapDir = offset >= 0 ? 1 : -1;
124 stepped = beyondThreshold;
125 } else {
126 state_.baseNs = current; // continuity; the residual offset is slewed in
127 state_.pendingOffsetNs = offset;
128 state_.slewEnd = now + std::chrono::duration_cast<std::chrono::steady_clock::duration>(config_.slewWindow);
129 state_.leap = false;
130 state_.leapDir = 0;
131 }
132 }
133 state_.everSynced = true;
134 state_.localAtSync = now;
135 state_.lastUpdate = now;
136 state_.rate = rate;
137 if (rate != 0.0) state_.rateCorrected = true;
138 state_.gateway = gateway;
139 state_.user = user;
140 snap = SnapshotLocked(now);
141 }
142 bool const statusChanged = (snap.status != lastNotified_);
143 Deliver(snap, stepped ? ChangeReason::kTimeStep : ChangeReason::kUpdate, statusChanged);
144 return ara::core::Result<void>();
145}

190ara::core::Result<void> TimeBaseResource::SetRateCorrection(double correction) {
191 if (!config_.allowProviderRateCorrection) {
192 return ara::core::Result<void>::FromError(MakeErrorCode(TsyncErrc::kFunctionNotSupported));
193 }
194 if (!std::isfinite(correction) || std::fabs(correction) > config_.providerRateDeviationMax ||
195 correction <= -1.0) {
196 return ara::core::Result<void>::FromError(MakeErrorCode(TsyncErrc::kLimitsExceeded));
197 }
198 auto const now = Now();
199 std::lock_guard<std::mutex> lk(mutex_);
200 if (state_.everSynced) {
201 // Re-base so the time stays continuous, then run at the new rate. The not yet applied part
202 // of a slew continues from "now".
203 std::int64_t remaining = 0;
204 if (state_.pendingOffsetNs != 0 && now < state_.slewEnd) {
205 std::int64_t const win = Ns(std::chrono::duration_cast<Duration>(state_.slewEnd - state_.localAtSync));
206 std::int64_t const dt = Ns(std::chrono::duration_cast<Duration>(now - state_.localAtSync));
207 std::int64_t const applied = win > 0 ? static_cast<std::int64_t>(std::llround(
208 static_cast<long double>(state_.pendingOffsetNs) * dt / win))
209 : state_.pendingOffsetNs;
210 remaining = state_.pendingOffsetNs - applied;
211 }
212 state_.baseNs = TimeAtLocked(state_, now);
213 state_.localAtSync = now;
214 state_.pendingOffsetNs = remaining;
215 }
216 state_.rate = correction;
217 state_.rateCorrected = true;
218 return ara::core::Result<void>();
219}

256ara::core::Result<void> TimeBaseResource::ReportPrecisionMeasurement(Duration pathDelay) {
257 auto const now = Now();
258 TimePrecisionMeasurement m;
259 std::vector<PrecisionCallback> cbs;
260 {
261 std::lock_guard<std::mutex> lk(mutex_);
262 if (!state_.everSynced) {
263 return ara::core::Result<void>::FromError(MakeErrorCode(TsyncErrc::kNotSynchronized));
264 }
265 std::int64_t const glb = TimeAtLocked(state_, now);
266 m.glbSeconds = static_cast<std::uint64_t>(glb / 1000000000LL);
267 m.glbNanoSeconds = static_cast<std::uint32_t>(glb % 1000000000LL);
268 m.timeBaseStatus = StatusLocked(state_, now).Bits();
269 if (!config_.isOffsetTimeBase) { // SWS_TS_00801: offset time bases record only the global time + status
270 std::int64_t const loc = std::chrono::duration_cast<Duration>(now.time_since_epoch()).count();
271 m.virtualLocalTimeLow = static_cast<std::uint32_t>(static_cast<std::uint64_t>(loc) & 0xFFFFFFFFULL);
272 m.rateDeviation = state_.rate;
273 m.locSeconds = static_cast<std::uint64_t>(loc / 1000000000LL);
274 m.locNanoSeconds = static_cast<std::uint32_t>(loc % 1000000000LL);
275 std::int64_t const pd = pathDelay.count();
276 m.pathDelay = pd < 0 ? 0U : static_cast<std::uint32_t>(pd > 0xFFFFFFFFLL ? 0xFFFFFFFFLL : pd);
277 }
278 for (auto const &o : precisionObservers_) cbs.push_back(o.second);
279 }
280 for (auto const &cb : cbs) {
281 if (cb) cb(m);
282 }
283 return ara::core::Result<void>();
