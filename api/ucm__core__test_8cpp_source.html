19PackageManagement::TransferIdType MakeId(std::uint8_t seed) {
20 PackageManagement::TransferIdType id{};
21 for (std::size_t i = 0; i < id.size(); ++i) id[i] = static_cast<std::uint8_t>(seed + i);
22 return id;
23}

25// Transfers `size` bytes in blocks of 1024 through the core; returns the id.
26PackageManagement::TransferIdType Upload(PackageManagerCore &c, std::uint64_t size,
27 std::uint8_t seed = 1, bool padLast = false) {
28 auto id = MakeId(seed);
29 auto s = c.TransferStart(size, id);
30 EXPECT_EQ(s.error, 0);
31 std::uint64_t sent = 0, block = 0;
32 while (sent < size) {
33 std::uint64_t n = std::min<std::uint64_t>(s.blockSize, size - sent);
34 auto r = c.TransferData(id, padLast ? s.blockSize : n, block++);
35 EXPECT_EQ(r.error, 0);
36 EXPECT_EQ(r.accepted, n);
37 sent += n;
38 }
39 EXPECT_EQ(c.TransferExit(id), 0);
40 return id;
41}

328TEST(UcmThreading, ConcurrentTransfersDoNotInterfere) {
329 PackageManagerCore c;
330 std::atomic<int> bad{0};
331 std::vector<std::thread> ts;
332 for (int t = 0; t < 8; ++t)
333 ts.emplace_back([&, t] {
334 auto id = MakeId(static_cast<std::uint8_t>(t * 20));
335 if (c.TransferStart(10 * 1024, id).error) ++bad;
336 for (std::uint64_t b = 0; b < 10; ++b)
337 if (c.TransferData(id, 1024, b).error) ++bad;
338 if (c.TransferExit(id)) ++bad;
339 });
340 for (auto &t : ts) t.join();
341 EXPECT_EQ(bad.load(), 0);
342}
