31bool ParseLevel(std::string v, LogLevel &out) {
32 std::transform(v.begin(), v.end(), v.begin(),
33 [](unsigned char c) { return static_cast<char>(std::tolower(c)); });
34 static constexpr struct { char const *n; LogLevel l; } kMap[] = {
35 {"off", LogLevel::kOff}, {"fatal", LogLevel::kFatal}, {"error", LogLevel::kError},
36 {"warn", LogLevel::kWarn}, {"info", LogLevel::kInfo}, {"debug", LogLevel::kDebug},
37 {"verbose", LogLevel::kVerbose}};
38 for (auto const &e : kMap)
39 if (v == e.n) { out = e.l; return true; }

76 static char const *d = "0123456789ABCDEF";
77 std::string s(digits, '0');
78 for (unsigned i = digits; i > 0; --i, v >>= 4) s[i - 1] = d[v & 0xF];

87std::string FormatInteger(std::uint64_t magnitude, bool negative, unsigned bytes, Format f) {
88 std::uint64_t const mask = bytes >= 8 ? ~std::uint64_t{0} : ((std::uint64_t{1} << (bytes * 8)) - 1);
89 auto const pad = [&](std::string digits) {
90 if (digits.size() < f.precision) digits.insert(0, f.precision - digits.size(), '0');
91 return digits;
92 };
93 auto const radix = [&](unsigned shift, char const *alphabet) {
94 std::uint64_t v = (negative ? (~magnitude + 1U) : magnitude) & mask;
95 std::string d;
96 do {
97 d.insert(d.begin(), alphabet[v & ((1U << shift) - 1U)]);
98 v >>= shift;
99 } while (v != 0U);
100 return pad(std::move(d));
101 };
102 switch (f.fmt) {
103 case Fmt::kHex: return "0x" + radix(4, "0123456789ABCDEF");
104 case Fmt::kBin: return "0b" + radix(1, "01");
105 case Fmt::kOct: return "0o" + radix(3, "01234567");
106 default: return (negative ? "-" : "") + pad(std::to_string(magnitude));
107 }
108}

110std::string FormatFloating(double v, Format f) {
111 char buf[512];
112 int const prec = f.precision == 0 ? 6 : std::min(static_cast<int>(f.precision), 100);
113 switch (f.fmt) {
114 case Fmt::kEngFloat: std::snprintf(buf, sizeof buf, "%.*e", prec, v); break;
115 case Fmt::kHexFloat:
116 if (f.precision == 0) std::snprintf(buf, sizeof buf, "%a", v);
117 else std::snprintf(buf, sizeof buf, "%.*a", prec, v);

119 case Fmt::kAutoFloat: std::snprintf(buf, sizeof buf, "%.*g", prec, v); break;
120 default: std::snprintf(buf, sizeof buf, "%.*f", prec, v); break;
121 }

123}

179Logger &CreateLogger(std::string_view ctxId, std::string_view ctxDescription,
180 LogLevel ctxDefLogLevel) {
181 auto &s = S();
182 std::lock_guard<std::mutex> lk(s.mutex);
183 std::string key(ctxId.substr(0, 4));
184 auto it = s.loggers.find(key);
185 if (it != s.loggers.end()) return *it->second;
186 if (!s.initialized) {
187 // Implicit default configuration: console sink, level from the first logger.
188 s.initialized = true;
189 s.defaultLevel = ctxDefLogLevel;
190 if (char const *lv = std::getenv("ARA_LOG_LEVEL")) ParseLevel(lv, s.defaultLevel);
191 s.sinks.push_back(std::make_shared<ConsoleSink>());
192 }
193 LogLevel lvl = ctxDefLogLevel;
194 if (char const *lv = std::getenv("ARA_LOG_LEVEL")) ParseLevel(lv, lvl);
195 auto *raw = new Logger(key, std::string(ctxDescription), lvl);
196 s.loggers.emplace(key, std::unique_ptr<Logger>(raw));
197 return *raw;
198}
