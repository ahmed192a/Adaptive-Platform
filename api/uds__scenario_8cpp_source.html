101std::string HexString(ByteVector const &b) {
102 std::string s;

104 for (std::size_t i = 0; i < b.size(); ++i) {

112std::optional<ByteVector> ParseHex(std::string const &text) {
113 std::string digits;
114 for (char c : text) {
115 if (std::isspace(static_cast<unsigned char>(c)) || c == ',') continue;
116 if (!std::isxdigit(static_cast<unsigned char>(c))) return std::nullopt;
117 digits += c;
118 }
119 if (digits.empty() || digits.size() % 2 != 0) return std::nullopt;
120 ByteVector out;
121 for (std::size_t i = 0; i < digits.size(); i += 2)
122 out.push_back(static_cast<std::uint8_t>(std::stoi(digits.substr(i, 2), nullptr, 16)));
123 return out;
124}

126std::string Describe(ByteVector const &m) {
127 char buf[96];
128 if (m.empty()) return "empty";
129 if (m[0] == 0x7F && m.size() == 3) {
130 std::snprintf(buf, sizeof buf, "negative response: %s (0x%02X) for %s", NrcName(m[2]), m[2], SidName(m[1]));
131 return buf;
132 }
133 if (m[0] >= 0x40 && m[0] < 0x7F) {
134 std::snprintf(buf, sizeof buf, "positive response: %s", SidName(m[0] - 0x40U));
135 return buf;
136 }
137 std::snprintf(buf, sizeof buf, "%s (0x%02X) request", SidName(m[0]), m[0]);
138 return buf;
139}
