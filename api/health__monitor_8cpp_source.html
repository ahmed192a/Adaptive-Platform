30Result<void> HealthMonitor::AddSupervision(std::shared_ptr<ElementarySupervision> s) {
31 std::lock_guard<std::mutex> lock(mMutex);
32 if (!s || mSupervisions.count(s->Name()) != 0) {
33 return Result<void>(MakeErrorCode(PhmErrc::kInvalidConfiguration, 0, "duplicate or null supervision"));
34 }
35 for (auto const &cp : s->Checkpoints()) {
36 mIndex[cp].push_back(s.get());
37 mKnownEntities.insert(cp.entity);
38 }
39 mSupervisions[s->Name()] = std::move(s);
40 return Result<void>();
41}

43Result<void> HealthMonitor::AddGlobalSupervision(GlobalSupervisionConfig cfg) {
44 std::lock_guard<std::mutex> lock(mMutex);
45 if (mGlobals.count(cfg.name) != 0 || cfg.members.empty()) {
46 return Result<void>(MakeErrorCode(PhmErrc::kInvalidConfiguration, 0, "duplicate global or no members"));
47 }
48 for (auto const &m : cfg.members) {
49 if (mSupervisions.count(m) == 0) {
50 return Result<void>(MakeErrorCode(PhmErrc::kUnknownSupervision, 0, "member supervision not configured"));
51 }
52 }
53 Global g;

213void HealthMonitor::EvaluateLocked(TimePoint now, std::vector<std::function<void()>> &deferred) {
214 for (auto &[name, s] : mSupervisions) {
215 if (!Suspended(*s)) {
216 s->Evaluate(now);
217 }
218 }
219 for (auto &[name, g] : mGlobals) {
220 GlobalSupervisionStatus const before = g.status;
221 if (before == GlobalSupervisionStatus::kDEACTIVATED) {
222 continue;
223 }
224 // aggregate: worst member status wins
225 bool anyExpired = false, anyFailed = false;
226 ElementarySupervision const *firstExpired = nullptr;
227 for (auto const &m : g.cfg.members) {
228 auto const &s = *mSupervisions[m];
229 if (s.Status() == ElementarySupervisionStatus::kEXPIRED) {
230 anyExpired = true;
231 if (firstExpired == nullptr) firstExpired = &s;
232 } else if (s.Status() == ElementarySupervisionStatus::kFAILED) {
233 anyFailed = true;
234 }
235 }
236 GlobalSupervisionStatus next = before;
237 if (before != GlobalSupervisionStatus::kSTOPPED) { // STOPPED is final
238 next = anyExpired ? GlobalSupervisionStatus::kEXPIRED
239 : anyFailed ? GlobalSupervisionStatus::kFAILED : GlobalSupervisionStatus::kOK;
240 }
241 if (next == GlobalSupervisionStatus::kEXPIRED && before != GlobalSupervisionStatus::kEXPIRED) {
242 g.expiredSince = now;
243 }
244 // ---- recovery on the EXPIRED edge ---------------------------------------
245 if (next == GlobalSupervisionStatus::kEXPIRED && !g.recoveryDispatched && firstExpired != nullptr) {
246 g.recoveryDispatched = true;
247 ExpiredSupervision ev{g.cfg.name, firstExpired->Name(), g.cfg.functionGroup,
248 firstExpired->Type(), g.cfg.executionError};
249 auto handler = mRecovery.find(g.cfg.recoveryAction);
250 if (handler == mRecovery.end()) {
251 if (g.cfg.escalateToWatchdog) {
252 Escalate(deferred); // nobody can recover: let the watchdog act
253 }
254 } else {
255 // run outside the lock; the resulting future is polled on later evaluations
256 auto fn = handler->second;
257 auto *gp = &g;
258 auto holder = std::make_shared<std::optional<ara::core::Future<void>>>();
259 deferred.emplace_back([fn, ev, holder] { holder->emplace(fn(ev)); });
260 deferred.emplace_back([this, gp, holder] {
261 std::lock_guard<std::mutex> lock(mMutex);
262 auto found = std::find_if(mGlobals.begin(), mGlobals.end(),
263 [gp](auto const &e) { return &e.second == gp; });
264 if (found != mGlobals.end() && holder->has_value()) {
265 found->second.pendingRecovery = std::move(**holder);
266 }
267 });
268 }
269 }
270 // ---- pending recovery result ------------------------------------------------
271 if (g.pendingRecovery.has_value() && g.pendingRecovery->is_ready()) {
272 auto res = g.pendingRecovery->GetResult();
273 g.pendingRecovery.reset();
274 if (!res.HasValue() && g.cfg.escalateToWatchdog) {
275 Escalate(deferred); // e.g. kSMCanNotHandleRecovery
276 }
277 }
278 // ---- platform critical supervisions: EXPIRED -> STOPPED ---------------------
279 if (next == GlobalSupervisionStatus::kEXPIRED && g.cfg.platformCritical &&
280 now - g.expiredSince >= g.cfg.expiredTolerance) {
281 next = GlobalSupervisionStatus::kSTOPPED;
282 Escalate(deferred);
283 }
284 g.status = next;
285 if (next != before && mListener) {
286 auto listener = mListener;
287 std::string const gname = g.cfg.name;
288 deferred.emplace_back([listener, gname, before, next] { listener(gname, before, next); });
289 }
290 }
291 bool anyStopped = std::any_of(mGlobals.begin(), mGlobals.end(), [](auto const &e) {
292 return e.second.status == GlobalSupervisionStatus::kSTOPPED;
293 });
294 if (mWatchdog && !anyStopped && !mWatchdogFired) {
295 auto wd = mWatchdog;
296 deferred.emplace_back([wd] { wd->AliveNotification(); });
297 }
298}

300void HealthMonitor::Evaluate(TimePoint now) {
301 std::vector<std::function<void()>> deferred;
302 {
303 std::lock_guard<std::mutex> lock(mMutex);
304 EvaluateLocked(now, deferred);
305 }
306 for (auto &d : deferred) {
307 d(); // callbacks may call back into the monitor, hence outside the lock
308 }

326 StopBackgroundEvaluation();
327 {
328 std::lock_guard<std::mutex> lock(mThreadMutex);
329 mThreadStop = false;
330 }
331 mThread = std::thread([this, period] {
332 std::unique_lock<std::mutex> lock(mThreadMutex);
333 while (!mThreadStop) {
334 lock.unlock();
335 Evaluate(ara::core::SteadyClock::now());

337 mThreadCv.wait_for(lock, period, [this] { return mThreadStop; });
