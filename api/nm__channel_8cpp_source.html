15Result<std::unique_ptr<NmChannel>> NmChannel::Create(std::string name, NmConfig cfg,
16 std::shared_ptr<NmClock> clock) {
17 using R = Result<std::unique_ptr<NmChannel>>;
18 if (!clock || cfg.repeatMessageTime.count() <= 0 || cfg.nmTimeout.count() <= 0 ||
19 cfg.waitBusSleepTime.count() <= 0 || cfg.messageCycleTime.count() <= 0 ||
20 cfg.pnResetTime.count() <= 0) {
21 return R::FromError(MakeErrorCode(NmErrc::kInvalidConfig));
22 }
23 // SWS_ANM_00056: the PN reset time must be longer than the NM message cycle time.
24 if (cfg.pnFilterMask != 0 && cfg.pnResetTime <= cfg.messageCycleTime) {
25 return R::FromError(MakeErrorCode(NmErrc::kInvalidConfig));
26 }
27 if (cfg.immediateNmTransmissions != 0 && cfg.immediateNmCycleTime.count() <= 0) {
28 return R::FromError(MakeErrorCode(NmErrc::kInvalidConfig));
29 }
30 if (cfg.messageCycleOffset.count() < 0) return R::FromError(MakeErrorCode(NmErrc::kInvalidConfig));
31 // The PDU layout must be consistent and able to carry the served partial networks.
32 if (!cfg.pduLayout.IsValid()) return R::FromError(MakeErrorCode(NmErrc::kInvalidConfig));
33 if (cfg.pnFilterMask != 0) {
34 std::size_t const bits = 8U * cfg.pduLayout.pnVectorLength;
35 if (bits == 0 || (bits < 32 && (cfg.pnFilterMask >> bits) != 0)) {
36 return R::FromError(MakeErrorCode(NmErrc::kInvalidConfig));
37 }
38 }
39 return R::FromValue(
40 std::unique_ptr<NmChannel>(new NmChannel(std::move(name), cfg, std::move(clock))));
41}

66void NmChannel::Enter(NmState s, Millis t, Pending &p) {
67 if (s == state_) return;
68 p.changes.emplace_back(state_, s);
69 state_ = s;
70 switch (s) {
71 case NmState::kRepeatMessage:
72 repeatDeadline_ = t + cfg_.repeatMessageTime;
73 nmTimeoutDeadline_ = t + cfg_.nmTimeout;
74 // nmMsgCycleOffset delays the first message; immediate transmissions start at once.
75 txDeadline_ = cfg_.immediateNmTransmissions > 0 ? t : t + cfg_.messageCycleOffset;
76 immediateRemaining_ = cfg_.immediateNmTransmissions;
77 break;
78 case NmState::kNormalOperation:
79 nmTimeoutDeadline_ = t + cfg_.nmTimeout;
80 break;
81 case NmState::kReadySleep: nmTimeoutDeadline_ = t + cfg_.nmTimeout; break;
82 case NmState::kPrepareBusSleep: sleepDeadline_ = t + cfg_.waitBusSleepTime; break;
83 case NmState::kBusSleep: break;
84 }
85}

87// Executes at most one timer-driven transition; returns true if one happened.
88bool NmChannel::Step(Millis now, Pending &p) {
89 switch (state_) {
90 case NmState::kRepeatMessage:
91 if (now >= repeatDeadline_) {
92 Enter(requested_ ? NmState::kNormalOperation : NmState::kReadySleep, repeatDeadline_, p);
93 return true;
94 }
95 return false;
96 case NmState::kReadySleep:
97 if (now >= nmTimeoutDeadline_) {
98 Enter(NmState::kPrepareBusSleep, nmTimeoutDeadline_, p);
99 return true;
100 }
101 return false;
102 case NmState::kPrepareBusSleep:
103 if (now >= sleepDeadline_) {
104 Enter(NmState::kBusSleep, sleepDeadline_, p);
105 return true;
106 }
107 return false;
108 case NmState::kNormalOperation:
109 case NmState::kBusSleep: return false;
110 }
111 return false;
112}

123void NmChannel::Pump(Millis now, Pending &p) {
124 while (Step(now, p)) {
125 }
126 for (std::uint8_t i = 0; i < kMaxPartialNetworks; ++i) {
127 std::uint32_t const bit = 1U << i;
128 if ((remotePn_ & bit) != 0 && now >= remotePnExpiry_[i]) remotePn_ &= ~bit;
129 }
130 if (Transmits(state_) && now >= txDeadline_) {
131 p.tx.push_back(MakePdu());
132 nmTimeoutDeadline_ = now + cfg_.nmTimeout; // SWS_ANM_00007..09: restarted on NM Tx as well as Rx
133 if (immediateRemaining_ > 0) {
134 --immediateRemaining_;
135 txDeadline_ = now + cfg_.immediateNmCycleTime;
136 } else {
137 txDeadline_ = now + cfg_.messageCycleTime;
138 }
139 }

142void NmChannel::Dispatch(Pending &p) {
143 StateHandler sh;
144 TxHandler th;
145 std::vector<std::pair<std::size_t, StateHandler>> so;
146 std::vector<std::pair<std::size_t, RequestedHandler>> ro;
147 {
148 std::lock_guard<std::mutex> l(mu_);
149 sh = onState_;
150 th = onTx_;
151 so = stateObservers_;
152 ro = requestedObservers_;
153 }
154 for (auto const &c : p.changes) {
155 if (sh) sh(c.first, c.second);
156 for (auto const &o : so) o.second(c.first, c.second);
157 }
158 for (bool r : p.requestedChanges) {
159 for (auto const &o : ro) o.second(r);
160 }
161 if (th) for (auto const &m : p.tx) th(m);
162}

164Result<void> NmChannel::Request() {
165 Pending p;
166 {
167 std::lock_guard<std::mutex> l(mu_);
168 Millis const now = clock_->Now();
169 Pump(now, p); // bring timers up to date before applying the request
170 if (!requested_) p.requestedChanges.push_back(true);
171 requested_ = true;
172 switch (state_) {
173 case NmState::kBusSleep:
174 case NmState::kPrepareBusSleep: Enter(NmState::kRepeatMessage, now, p); break;
175 case NmState::kReadySleep:
176 Enter(NmState::kNormalOperation, now, p);
177 txDeadline_ = now;
178 break;
179 case NmState::kRepeatMessage:
180 case NmState::kNormalOperation: break;
181 }
182 Pump(now, p);
183 }
184 Dispatch(p);
185 return Result<void>::FromValue();
186}

231void NmChannel::OnPduReceived(NmPdu const &pdu) {
232 Pending p;
233 {
234 std::lock_guard<std::mutex> l(mu_);
235 Millis const now = clock_->Now();
236 Pump(now, p);
237 // SWS_ANM_00081/00089: with partial networking active, drop PNI=1 messages that carry no
238 // bit of a partial network this node serves (unless allNmMessagesKeepAwake).
239 bool const dropped = cfg_.pnFilterMask != 0 && pdu.partialNetworkInfo &&
240 !cfg_.allNmMessagesKeepAwake && (pdu.pnRequest & cfg_.pnFilterMask) == 0;
241 if (!dropped) {
242 nmTimeoutDeadline_ = now + cfg_.nmTimeout;
243 if (pdu.partialNetworkInfo) {
244 std::uint32_t const bits = pdu.pnRequest & cfg_.pnFilterMask;
245 for (std::uint8_t i = 0; i < kMaxPartialNetworks; ++i) {
246 if ((bits & (1U << i)) != 0) {
247 remotePn_ |= 1U << i;
248 remotePnExpiry_[i] = now + cfg_.pnResetTime;
249 }
250 }
251 }
252 switch (state_) {
253 case NmState::kBusSleep:
254 case NmState::kPrepareBusSleep: Enter(NmState::kRepeatMessage, now, p); break;
255 case NmState::kReadySleep:
256 case NmState::kNormalOperation:
257 if (pdu.repeatMessageRequest) Enter(NmState::kRepeatMessage, now, p);
258 break;
259 case NmState::kRepeatMessage: break;
260 }
261 }
262 Pump(now, p);
263 }
264 Dispatch(p);
265}

287void NmChannel::Shutdown() {
288 Pending p;
289 {
290 std::lock_guard<std::mutex> l(mu_);
291 Millis const now = clock_->Now();
292 Pump(now, p);
293 localPn_ = 0;
294 if (requested_) {
295 requested_ = false;
296 p.requestedChanges.push_back(false);
297 }
298 // SWS_ANM_00090: afterwards the network state reads kNoCom; the remaining timers run down
299 // from Prepare Bus-Sleep to Bus-Sleep.
300 if (state_ == NmState::kRepeatMessage || state_ == NmState::kNormalOperation ||
301 state_ == NmState::kReadySleep) {
302 Enter(NmState::kPrepareBusSleep, now, p);
303 }
304 }
305 Dispatch(p);
306}
