43std::string HexOf(std::vector<std::uint8_t> const &v) {
44 static char const *d = "0123456789abcdef";
45 std::string s;
46 for (auto b : v) {
47 s += d[b >> 4];
48 s += d[b & 0xF];

52bool UnhexOf(std::string const &text, std::vector<std::uint8_t> &out) {
53 out.clear();
54 int hi = -1;
55 for (char c : text) {
56 if (std::isspace(static_cast<unsigned char>(c)) != 0) continue;
57 int v = (c >= '0' && c <= '9') ? c - '0' : (c >= 'a' && c <= 'f') ? c - 'a' + 10 : (c >= 'A' && c <= 'F') ? c - 'A' + 10 : -1;
58 if (v < 0) return false;
59 if (hi < 0) {
60 hi = v;
61 } else {
62 out.push_back(static_cast<std::uint8_t>((hi << 4) | v));
63 hi = -1;
64 }
65 }
66 return hi < 0;
67}

562 int exitCode{-1};
563 bool timedOut{false};
564 bool truncated{false};
565 std::string output;
566};
567
569RunResult RunProcess(fs::path const &exe, std::vector<std::string> const &args, fs::path const &cwd, int timeoutMs) {
570 RunResult res;
571 int pipefd[2];
572 if (::pipe(pipefd) != 0) {
573 res.output = "pipe failed";
574 return res;
575 }
576 std::vector<std::string> store{exe.string()};
577 store.insert(store.end(), args.begin(), args.end());
578 std::vector<char *> argv;
579 for (auto &s : store) argv.push_back(s.data());
580 argv.push_back(nullptr);
581 std::string const cwdStr = cwd.string();
582
583 pid_t pid = ::fork();
584 if (pid < 0) {
585 ::close(pipefd[0]);
586 ::close(pipefd[1]);
587 res.output = "fork failed";
588 return res;
589 }
590 if (pid == 0) {
591 ::setsid();
592 int devnull = ::open("/dev/null", O_RDONLY);
593 if (devnull >= 0) ::dup2(devnull, 0);
594 ::dup2(pipefd[1], 1);
595 ::dup2(pipefd[1], 2);
596 ::close(pipefd[0]);
597 ::close(pipefd[1]);
598 if (::chdir(cwdStr.c_str()) != 0) ::_exit(126);
599 ::execv(argv[0], argv.data());
600 ::_exit(127);
601 }
602 ::close(pipefd[1]);
603 constexpr std::size_t kMaxOut = 64 * 1024;
604 auto const deadline = std::chrono::steady_clock::now() + std::chrono::milliseconds(timeoutMs);
605 for (;;) {
606 auto left = std::chrono::duration_cast<std::chrono::milliseconds>(deadline - std::chrono::steady_clock::now()).count();
607 if (left <= 0) {
608 res.timedOut = true;
609 break;
610 }
611 pollfd pfd{pipefd[0], POLLIN, 0};
612 int pr = ::poll(&pfd, 1, static_cast<int>(std::min<long long>(left, 200)));
613 if (pr < 0 && errno != EINTR) break;
614 if (pr > 0) {
615 char buf[4096];
616 ssize_t n = ::read(pipefd[0], buf, sizeof buf);
617 if (n <= 0) break;
618 if (res.output.size() < kMaxOut) res.output.append(buf, static_cast<std::size_t>(n));
619 else res.truncated = true;
620 }
621 }
622 ::close(pipefd[0]);
623 if (res.timedOut) ::kill(-pid, SIGKILL);
624 int status = 0;
625 ::waitpid(pid, &status, 0);
626 if (WIFEXITED(status)) res.exitCode = WEXITSTATUS(status);
627 else if (WIFSIGNALED(status)) res.exitCode = 128 + WTERMSIG(status);
628 if (res.output.size() > kMaxOut) {
629 res.output.resize(kMaxOut);
630 res.truncated = true;
631 }
632 return res;
633}

669HttpResponse Features::Handle(std::string const &path, json const &in) {
670 auto slash = path.find('/', 5);
671 std::string const area = path.substr(5, slash - 5);
672 std::string const op = slash == std::string::npos ? "" : path.substr(slash + 1);
673 Impl &m = *impl_;
674
675 if (area == "apps") {
676 if (op != "run") return Err("unknown apps operation", 404);
677 std::string const name = in.value("name", "");
678 std::string const mode = in.value("mode", "");
679 for (auto const &spec : Apps()) {
680 if (name != spec.name) continue;
681 for (auto const &md : spec.modes) {
682 if (md.first != mode) continue;
683 fs::path exe = m.opt.demoDir / spec.name / spec.name;
684 std::error_code ec;
685 if (!fs::exists(exe, ec) || ::access(exe.c_str(), X_OK) != 0) return Err("demo binary not built: " + exe.string());
686 std::unique_lock<std::mutex> lk(m.appMutex, std::try_to_lock);
687 if (!lk.owns_lock()) return Err("another demo app is still running", 409);
688 fs::path work = m.opt.dataDir / "apps";
689 fs::create_directories(work, ec);
690 std::vector<std::string> args;
691 for (auto a : md.second) {
692 if (a.rfind("@DATA@", 0) == 0) {
693 a = (work / a.substr(7)).string();
694 fs::remove_all(a, ec); // every run starts from a clean slate
695 }
696 args.push_back(std::move(a));
697 }
698 auto t0 = std::chrono::steady_clock::now();
699 RunResult r = RunProcess(exe, args, work, 40000);
700 json j = {{"ok", true}, {"name", name}, {"mode", mode}, {"exit", r.exitCode}, {"timedOut", r.timedOut},
701 {"truncated", r.truncated}, {"output", r.output},
702 {"ms", std::chrono::duration_cast<std::chrono::milliseconds>(std::chrono::steady_clock::now() - t0).count()}};
703 std::lock_guard<std::mutex> g(m.mutex);
704 m.lastApp = j;
705 return Json(j);
706 }
707 return Err("unknown mode", 404);
708 }
709 return Err("unknown demo app", 404);
710 }
711
712 std::lock_guard<std::mutex> g(m.mutex);
713 if (area == "nm") return m.nm.Handle(op, in);
714 if (area == "tsync") return m.tsync.Handle(op, in);
715 if (area == "udsx") return m.udsx.Handle(op, in);
716 if (area == "update") return m.update.Handle(op, in);
717 if (area == "keys") {
718#ifdef AP_NODE_CRYPTO
719 return m.keys.Handle(op, in);
720#else
721 return Err("built without crypto support");
722#endif
723 }
724 return Err("not found", 404);
725}
