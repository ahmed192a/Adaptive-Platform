144TEST_F(Spec, NotifiersFireForStatusSyncStateAndLeap) { // SWS_TS_00701 / 00702 / 00703
145 SynchronizedTimeBaseProvider p(port);
146 SynchronizedTimeBaseConsumer c(port);
147 std::vector<SynchronizationStatus> syncs;
148 std::vector<LeapJump> leaps;
149 int statusCalls = 0;
150 ASSERT_TRUE(c.RegisterStatusChangeNotifier([&](SynchronizedTimeBaseStatus const &) { ++statusCalls; }).HasValue());
151 ASSERT_TRUE(c.RegisterSynchronizationStateChangeNotifier([&](SynchronizationStatus s) { syncs.push_back(s); }).HasValue());
152 ASSERT_TRUE(c.RegisterTimeLeapNotifier([&](LeapJump l) { leaps.push_back(l); }).HasValue());
153 p.SetTime(T(seconds(1))); // -> synchronized
154 p.UpdateTime(T(seconds(30))); // leap into the future
155 Advance(seconds(2));
156 tb->Poll(); // timeout detected
157 ASSERT_GE(syncs.size(), 2U);
158 EXPECT_EQ(syncs.front(), SynchronizationStatus::kSynchronized);
159 EXPECT_EQ(syncs.back(), SynchronizationStatus::kTimeOut);
160 ASSERT_EQ(leaps.size(), 1U);
161 EXPECT_EQ(leaps[0], LeapJump::kTimeLeapFuture);
162 EXPECT_GE(statusCalls, 2);
163
164 c.UnregisterTimeLeapNotifier();
165 p.UpdateTime(T(seconds(500)));
166 EXPECT_EQ(leaps.size(), 1U);
167}
