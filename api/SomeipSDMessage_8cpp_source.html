52uint32_t SomeIpSDMessage::Length() const noexcept {
53 // [PRS_SOMEIPSD_00153]: bytes after the Length field. That is the rest of
54 // the SOME/IP header (8), Flags + Reserved (4), the entries array length
55 // field (4) + entries, and the options array length field (4) + options.
56 const uint32_t LengthFieldSize = 4;
57 const uint32_t SdHeaderSize = 4;
58
59 return EmptyLength + SdHeaderSize + LengthFieldSize + getEntriesLength() +
60 LengthFieldSize + getOptionsLength();
61}

80std::vector<uint8_t> SomeIpSDMessage::Serializer() {
81
82 // General SOME/IP header payload insertion
83 std::vector<uint8_t> _result = Message::Serializer();
84
85 // Flags byte: Reboot (0x80) and Unicast (0x40, always 1)
86 // [PRS_SOMEIPSD_00254, PRS_SOMEIPSD_00259]; all other bits (incl. the
87 // Explicit Initial Data Control bit removed in R21-11) and the 24 reserved
88 // bits are zero [PRS_SOMEIPSD_00702, PRS_SOMEIPSD_00261].
89 const uint32_t RebootedFlag = 0xc0000000;
90 const uint32_t NotRebootedFlag = 0x40000000;
91 helper::Inject(_result, Rebooted ? RebootedFlag : NotRebootedFlag);
92 uint8_t lastOptionIndex = 0;
93 std::vector<uint8_t> entriesPayload;
94 std::vector<uint8_t> optionsPayload;
95 for (auto const entry : entries) {
96
97 auto entryPayload = entry->Payload(lastOptionIndex);
98 helper::Concat(entriesPayload, std::move(entryPayload));
99
100 for (auto firstOption : entry->FirstOptions()) {
101
102 auto firstOptionPayload = firstOption->Payload();
103 helper::Concat(optionsPayload, std::move(firstOptionPayload));
104 }
105
106 for (auto secondOption : entry->SecondOptions()) {
107
108 auto secondOptionPayload = secondOption->Payload();
109 helper::Concat(optionsPayload, std::move(secondOptionPayload));
110 }
111 }
112
113 // Entries length and payloads insertion
114 uint32_t entriesLength = getEntriesLength();
115 helper::Inject(_result, entriesLength);
116 helper::Concat(_result, std::move(entriesPayload));
117
118 // Options length and payloads insertion
119 uint32_t optionsLength = getOptionsLength();
120 helper::Inject(_result, optionsLength);
121 helper::Concat(_result, std::move(optionsPayload));
122
123 // The base class wrote the payload-less Length; patch in the real one.
124 const uint32_t length = Length();
125 _result[4] = static_cast<uint8_t>(length >> 24);
126 _result[5] = static_cast<uint8_t>(length >> 16);
127 _result[6] = static_cast<uint8_t>(length >> 8);
128 _result[7] = static_cast<uint8_t>(length);
129
130 return _result;
131}

141bool SomeIpSDMessage::Deserialize(const std::vector<uint8_t> &payload) {
142 const size_t EntrySize = 16;
143 const size_t OptionHeaderSize = 3; // Length(2) + Type(1)
144 // 16 byte header + flags/reserved + entries length field
145 if (payload.size() < 16 + 4 + 4) {
146 return false;
147 }
148 size_t offset = 16;
149 // Flags are the first byte, the highest bit is the Reboot flag
150 // [PRS_SOMEIPSD_00254].
151 const bool rebooted = (payload[offset] & 0x80) != 0;
152 offset += 4;
153
154 const uint64_t entriesLength = Be32(payload, offset);
155 offset += 4;
156 if (entriesLength % EntrySize != 0 ||
157 offset + entriesLength + 4 > payload.size()) {
158 return false;
159 }
160 const size_t entriesStart = offset;
161 offset += static_cast<size_t>(entriesLength);
162
163 const uint64_t optionsLength = Be32(payload, offset);
164 offset += 4;
165 if (offset + optionsLength > payload.size()) {
166 return false;
167 }
168 const size_t optionsStart = offset;
169
170 // Options array: walk by the on-wire Length field so that option types we
171 // do not parse can still be skipped (their slot stays nullptr).
172 std::vector<std::shared_ptr<ara::com::option::Option>> options;
173 size_t optionsOffset = 0;
174 while (optionsOffset < optionsLength) {
175 if (optionsOffset + OptionHeaderSize > optionsLength) {
176 return false;
177 }
178 const size_t o = optionsStart + optionsOffset;
179 const size_t len = (static_cast<size_t>(payload[o]) << 8) | payload[o + 1];
180 if (optionsOffset + OptionHeaderSize + len > optionsLength) {
181 return false;
182 }
183 std::vector<uint8_t> optionBytes(payload.begin() + o,
184 payload.begin() + o + OptionHeaderSize + len);
185 std::shared_ptr<ara::com::option::Option> option;
186 // The option parsers read fixed layouts; only hand them complete ones.
187 if (len >= 9) {
188 option = option::OptionDeserializer::Deserialize(optionBytes, 0);
189 }
190 options.push_back(option);
191 optionsOffset += OptionHeaderSize + len;
192 }
193
194 this->Rebooted = rebooted;
195 for (size_t e = 0; e < entriesLength; e += EntrySize) {
196 const size_t o = entriesStart + e;
197 // Index of first/second option run, then #opt1 (high) / #opt2 (low).
198 const size_t firstIndex = payload[o + 1];
199 const size_t secondIndex = payload[o + 2];
200 const size_t firstCount = (payload[o + 3] >> 4) & 0x0f;
201 const size_t secondCount = payload[o + 3] & 0x0f;
202
203 std::vector<uint8_t> entryBytes(payload.begin() + o,
204 payload.begin() + o + EntrySize);
205 auto entry = entry::Entry_Deserializer::Deserialize(entryBytes, 0);
206 if (entry == nullptr) {
207 continue; // unknown entry type: ignore it
208 }
209 for (size_t i = 0; i < firstCount; ++i) {
210 if (firstIndex + i < options.size() && options[firstIndex + i]) {
211 entry->AddFirstOption(options[firstIndex + i]);
212 }
213 }
214 for (size_t i = 0; i < secondCount; ++i) {
215 if (secondIndex + i < options.size() && options[secondIndex + i]) {
216 entry->AddSecondOption(options[secondIndex + i]);
217 }
218 }
219 this->entries.push_back(entry);
220 }
221 return true;
222}
