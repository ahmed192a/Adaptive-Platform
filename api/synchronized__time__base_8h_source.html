43class SynchronizedTimeBaseStatus {

45 SynchronizedTimeBaseStatus() = default;
46 explicit SynchronizedTimeBaseStatus(TimeBaseSnapshot const &snap, bool synchronizedEver)
47 : snap_(snap), ever_(synchronizedEver) {}
48
49 SynchronizationStatus GetSynchronizationStatus() const noexcept {
50 return ever_ ? ToSynchronizationStatus(snap_.status) : SynchronizationStatus::kNotSynchronizedUntilStartup;
51 }
52 LeapJump GetLeapJump() const noexcept { return snap_.leapJump; }
54 std::vector<std::uint8_t> GetUserData() const {
55 return std::vector<std::uint8_t>(snap_.userData.bytes, snap_.userData.bytes + snap_.userData.length);
56 }
57 Timestamp GetCreationTime() const noexcept { return snap_.time; }
58 std::chrono::steady_clock::time_point GetCreationLocalTime() const noexcept { return snap_.localTime; }
60 bool GetRateCorrected() const noexcept { return snap_.rateCorrected; }
62 bool GetRateExceeded() const noexcept { return snap_.rateExceeded; }
63
64private:
