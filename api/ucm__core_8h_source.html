162 StartResult TransferStart(std::uint64_t size, PM::TransferIdType const &newId) {
163 std::lock_guard<std::mutex> g(m_);
164 StartResult r;
165 if (size == 0 || size > limits_.maxPackageBytes) {
166 r.error = PM::InsufficientMemory;
167 return r;
168 }
169 Package p;
170 p.size = size;
171 p.state = pkgstate::kTransferring;
172 packages_[newId] = p;
173 r.id = newId;
174 r.blockSize = limits_.blockSize;
175 return r;
176 }

181 DataResult TransferData(PM::TransferIdType const &id, std::uint64_t dataSize,
182 std::uint64_t blockCounter) {
183 std::lock_guard<std::mutex> g(m_);
184 DataResult r;
185 auto it = packages_.find(id);
186 if (it == packages_.end()) { r.error = PM::InvalidTransferId; return r; } // 00203
187 Package &p = it->second;
188 if (p.state != pkgstate::kTransferring) { // 00208
189 r.error = PM::OperationNotPermitted;
190 return r;
191 }
192 if (blockCounter != p.nextBlock) { r.error = PM::IncorrectBlock; return r; } // 00204
193 if (dataSize > limits_.blockSize) { r.error = PM::IncorrectBlockSize; return r; } // 00243
194 std::uint64_t const remaining = p.size - p.received;
195 if (remaining == 0) { r.error = PM::IncorrectSize; return r; } // 00205
196 r.accepted = std::min<std::uint64_t>(dataSize, remaining);
197 p.received += r.accepted;
198 ++p.nextBlock;
199 return r;
200 }

204 Error TransferExit(PM::TransferIdType const &id) {
205 std::lock_guard<std::mutex> g(m_);
206 auto it = packages_.find(id);
207 if (it == packages_.end()) return PM::InvalidTransferId;
208 Package &p = it->second;
209 if (p.state != pkgstate::kTransferring) return PM::OperationNotPermitted;
210 if (p.received < p.size) return PM::InsufficientData;
211 if (hooks_.validatePackage) { // 00294 / 00098 / 00092 / 00103: the package stays kTransferring
212 int const e = hooks_.validatePackage(id); // so the client can DeleteTransfer or retry.
213 if (e != kOk) return e;

215 p.state = pkgstate::kTransferred;
216 return kOk;

232 // -------------------------------------------------------------- processing
235 Error BeginProcess(PM::TransferIdType const &id) {
236 PM::PackageManagerStatusType s;
237 {
238 std::lock_guard<std::mutex> g(m_);
239 auto it = packages_.find(id);
240 if (it == packages_.end()) return PM::InvalidTransferId;
241 if (status_ == status::kProcessing) return PM::ServiceBusy;
242 if ((status_ != status::kIdle && status_ != status::kReady) ||
243 (it->second.state != pkgstate::kTransferred && it->second.state != pkgstate::kProcessingFailed))
244 return PM::OperationNotPermitted;
245 it->second.state = pkgstate::kProcessing;
246 it->second.cancel = false;
247 status_ = status::kProcessing;
248 s = status_;
249 }
250 Notify(s);
251 return kOk;

454 bool Load(std::string const &path) {
455 std::ifstream f(path);
456 std::string magic;
457 int version = 0;
458 if (!(f >> magic >> version) || magic != "ucm-state" || version != 1) return false;
459 std::string key;
460 unsigned st = 0;
461 if (!(f >> key >> st) || key != "status" || st > status::kVerifying) return false;
462 std::map<PM::TransferIdType, Package> pk;
463 std::string idHex;
464 while (f >> key) {
465 Package p;
466 unsigned ps = 0;
467 if (key != "pkg" || !(f >> idHex >> p.size >> p.received >> p.nextBlock >> ps) || idHex.size() != 32 ||
468 ps > pkgstate::kProcessingFailed)
469 return false;
470 PM::TransferIdType id{};
471 for (std::size_t i = 0; i < 16; ++i) {
472 char *end = nullptr;
473 std::string const byte = idHex.substr(i * 2, 2);
474 unsigned long v = std::strtoul(byte.c_str(), &end, 16);
475 if (end != byte.c_str() + 2) return false;
476 id[i] = static_cast<std::uint8_t>(v);
477 }
478 p.state = static_cast<PM::SwPackageStateType>(ps);
479 pk[id] = p;
480 }
481 PM::PackageManagerStatusType s = static_cast<PM::PackageManagerStatusType>(st);
482 {
483 std::lock_guard<std::mutex> g(m_);
484 packages_ = std::move(pk);
485 for (auto &kv : packages_)
486 if (kv.second.state == pkgstate::kProcessing || kv.second.state == pkgstate::kProcessingStream)
487 kv.second.state = pkgstate::kProcessingFailed;
488 switch (s) {
489 case status::kProcessing: s = AnyProcessed() ? status::kReady : status::kIdle; break;
490 case status::kActivating: s = status::kActivated; break;
491 case status::kRollingBack: s = status::kRolledBack; break;
492 case status::kCleaningUp: s = status::kIdle; break;
493 default: break;
494 }
495 status_ = s;
496 }
497 Notify(s);
498 return true;
499 }

510 void Record(std::string action, PM::TransferIdType const &id, int result) {
511 HistoryEntry e;
512 e.timeSeconds = clock_ ? clock_() : static_cast<std::uint64_t>(std::chrono::duration_cast<std::chrono::seconds>(
513 std::chrono::steady_clock::now().time_since_epoch()).count());
514 e.action = std::move(action);
515 e.transferId = id == PM::TransferIdType{} ? std::string() : IdToString(id);
516 e.result = result;
517 std::lock_guard<std::mutex> g(hm_);
518 e.sequence = history_.size() + 1;
519 history_.push_back(std::move(e));
520 }
