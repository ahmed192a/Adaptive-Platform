32// SWS_EM_02243: ReportExecutionState(kRunning) is only valid from the Starting state.
33TEST_F(SpecExec, SecondKRunningReportIsAnInvalidTransition) {
34 ASSERT_EQ(::mkfifo("execution_client_fifo", 0600), 0);
35 int reader = ::open("execution_client_fifo", O_RDONLY | O_NONBLOCK);
36 ASSERT_GE(reader, 0);
37 {
38 ExecutionClient ec;
39 auto first = ec.ReportExecutionState(ExecutionState::kRunning);
40 EXPECT_TRUE(std::holds_alternative<std::monostate>(first));
41 auto second = ec.ReportExecutionState(ExecutionState::kRunning);
42 ASSERT_TRUE(std::holds_alternative<ExecErrc>(second));
43 EXPECT_EQ(std::get<ExecErrc>(second), ExecErrc::kInvalidTransition);
44 // only the first report reached Execution Management
45 ExecutionState got{};
46 EXPECT_EQ(::read(reader, &got, sizeof(got)), static_cast<ssize_t>(sizeof(got)));
47 EXPECT_EQ(got, ExecutionState::kRunning);
48 EXPECT_EQ(::read(reader, &got, sizeof(got)), -1);
49 // terminating is still allowed afterwards
50 EXPECT_TRUE(std::holds_alternative<std::monostate>(ec.ReportExecutionState(ExecutionState::kTerminating)));
51 }
52 ::close(reader);
53}
