44std::string ToHex(std::vector<std::uint8_t> const &v) {
45 static char const *d = "0123456789ABCDEF";
46 std::string s;
47 for (std::size_t i = 0; i < v.size(); ++i) {
48 if (i != 0) s += ' ';
49 s += d[v[i] >> 4];
50 s += d[v[i] & 0xF];
51 }
52 return s;
53}
54bool FromHex(std::string const &text, std::vector<std::uint8_t> &out) {
55 out.clear();
56 int hi = -1;
57 for (char c : text) {
58 if (std::isspace(static_cast<unsigned char>(c))) continue;
59 int v = std::isdigit(static_cast<unsigned char>(c)) ? c - '0'
60 : (c >= 'a' && c <= 'f') ? c - 'a' + 10
61 : (c >= 'A' && c <= 'F') ? c - 'A' + 10 : -1;
62 if (v < 0) return false;
63 if (hi < 0) hi = v;
64 else {
65 out.push_back(static_cast<std::uint8_t>(hi * 16 + v));
66 hi = -1;
67 }
68 }
69 return hi < 0;
70}

413 HttpResponse Crypto(std::string const &op, json const &in) {
414#ifdef AP_NODE_CRYPTO
415 using namespace ara::crypto;
416 auto bytes = [](std::string const &s) { return ByteVector(s.begin(), s.end()); };
417 auto hexs = [](ByteVector const &v) { return ToHex(v); };
418 std::string text = in.value("text", "");
419 std::string aad = in.value("aad", "");
420 auto fail = [](auto const &r) { return Err(r.Error().Message(), 200); };
421 if (op == "hash") {
422 std::string alg = in.value("alg", "sha256");
423 HashAlgorithm a = alg == "sha512" ? HashAlgorithm::kSha512 : alg == "sha384" ? HashAlgorithm::kSha384 : HashAlgorithm::kSha256;
424 auto h = Hash(a, bytes(text));
425 if (!h.HasValue()) return fail(h);
426 return Json({{"ok", true}, {"hex", hexs(h.Value())}});
427 }
428 if (op == "hmac") {
429 auto m = MessageAuthnCode::Create(HashAlgorithm::kSha256, *macKey);
430 if (!m.HasValue()) return fail(m);
431 (void)m.Value().Update(bytes(text));
432 auto t = m.Value().Finish();
433 if (!t.HasValue()) return fail(t);
434 return Json({{"ok", true}, {"hex", hexs(t.Value())}});
435 }
436 if (op == "seal") {
437 auto s = AeadSeal(*aesKey, bytes(aad), bytes(text));
438 if (!s.HasValue()) return fail(s);
439 return Json({{"ok", true}, {"hex", hexs(s.Value())}});
440 }
441 if (op == "open") {
442 ByteVector sealed;
443 if (!FromHex(in.value("hex", ""), sealed)) return Err("not valid hex", 200);
444 auto o = AeadOpen(*aesKey, bytes(aad), sealed);
445 if (!o.HasValue()) return Json({{"ok", false}, {"error", o.Error().Message()}, {"authFailed", true}});
446 return Json({{"ok", true}, {"text", std::string(o.Value().begin(), o.Value().end())}});
447 }
448 if (op == "sign" || op == "verify") {
449 bool ed = in.value("alg", "ed25519") == "ed25519";
450 PrivateKey &pk = ed ? *edKey : *ecKey;
451 if (op == "sign") {
452 auto s = pk.Sign(bytes(text));
453 if (!s.HasValue()) return fail(s);
454 return Json({{"ok", true}, {"hex", hexs(s.Value())}});
455 }
456 ByteVector sig;
457 if (!FromHex(in.value("sig", ""), sig)) return Err("signature is not valid hex", 200);
458 auto pub = pk.GetPublicKey();
459 if (!pub.HasValue()) return fail(pub);
460 auto v = pub.Value().Verify(bytes(text), sig);
461 return Json({{"ok", true}, {"valid", v.HasValue() && v.Value()}});
462 }
463 return Err("unknown crypto operation", 404);
464#else
465 (void)op; (void)in;
466 return Err("built without ara::crypto", 501);
467#endif
468 }

481 json StartEm() {
482 if (EmRunning()) return {{"ok", false}, {"error", "already running"}};
483 fs::path exe = opt.platformDir / "Adaptive_Platform";

507 void StopEm() {
508 pid_t pid;
509 {
510 std::lock_guard<std::mutex> lk(emMutex);
511 pid = emPid;
512 emPid = -1;
513 }
514 if (pid <= 0) return;
515 ::kill(-pid, SIGTERM);
516 for (int i = 0; i < 30; ++i) {
517 int st = 0;
518 if (::waitpid(pid, &st, WNOHANG) == pid) break;
519 std::this_thread::sleep_for(100ms);
520 }
521 ::kill(-pid, SIGKILL); // leftovers of the process group
522 int st = 0;

526 json EmProcesses() {
527 json out = json::array();
528 pid_t pg;
529 {
530 std::lock_guard<std::mutex> lk(emMutex);
531 pg = emPid;
532 }
533 if (pg <= 0) return out;
534 std::error_code ec;
535 for (auto const &e : fs::directory_iterator("/proc", ec)) {
536 auto n = e.path().filename().string();
537 if (n.empty() || !std::isdigit(static_cast<unsigned char>(n[0]))) continue;
538 std::ifstream f(e.path() / "stat");
539 std::string line;
540 if (!std::getline(f, line)) continue;
541 auto lp = line.find('('), rp = line.rfind(')');
542 if (lp == std::string::npos || rp == std::string::npos) continue;
543 std::istringstream rest(line.substr(rp + 2));
544 std::string state;
545 int ppid = 0, pgrp = 0;
546 rest >> state >> ppid >> pgrp;
547 if (pgrp == pg && state != "Z") out.push_back({{"pid", std::stoi(n)}, {"name", line.substr(lp + 1, rp - lp - 1)}});
548 }
549 return out;
550 }
551 std::string ReadTail(fs::path const &p, std::size_t maxBytes) {
552 std::ifstream f(p, std::ios::binary);
553 if (!f) return {};
554 f.seekg(0, std::ios::end);
555 auto size = static_cast<std::size_t>(f.tellg());
556 std::size_t off = size > maxBytes ? size - maxBytes : 0;
557 f.seekg(static_cast<std::streamoff>(off));
558 std::string s(size - off, '\0');
559 f.read(s.data(), static_cast<std::streamsize>(s.size()));
560 return s;
561 }

564 HttpResponse Static(std::string path) {
565 if (path == "/") path = "/index.html";
566 if (path.find("..") != std::string::npos) return {403, "text/plain", "forbidden"};
567 fs::path p = opt.webRoot / path.substr(1);
568 std::ifstream f(p, std::ios::binary);
569 if (!f) return {404, "text/plain", "not found"};
570 f.seekg(0, std::ios::end);
571 auto const len = f.tellg();
572 std::string body;
573 if (len > 0) {
574 body.resize(static_cast<std::size_t>(len));
575 f.seekg(0, std::ios::beg);
576 f.read(body.data(), static_cast<std::streamsize>(body.size()));
577 body.resize(static_cast<std::size_t>(f.gcount()));
578 }
579 auto ext = p.extension().string();
580 std::string ct = ext == ".html" ? "text/html; charset=utf-8" : ext == ".css" ? "text/css"
581 : ext == ".js" ? "application/javascript" : ext == ".svg" ? "image/svg+xml" : "application/octet-stream";
582 return {200, ct, std::move(body)};
583 }

585 HttpResponse Handle(HttpRequest const &req) {
586 if (req.path.rfind("/api/", 0) != 0) return req.method == "GET" ? Static(req.path) : HttpResponse{405, "text/plain", "method not allowed"};
587 json in = json::object();
588 if (req.method == "POST" && !req.body.empty()) {
589 in = json::parse(req.body, nullptr, false);
590 if (in.is_discarded() || !in.is_object()) return Err("body must be a JSON object");
591 }
592 std::string const &p = req.path;
593
594 if (features && Features::Owns(p)) {
595 if (req.method != "POST") return {405, "text/plain", "method not allowed"};
596 return features->Handle(p, in);
597 }
598 if (p == "/api/overview") {
599 auto since = req.query.count("since") ? std::strtoull(req.query.at("since").c_str(), nullptr, 10) : 0ULL;
600 json j = {{"time", NowMs()},
601 {"phm", PhmSnapshot()},
602 {"diag", {{"session", static_cast<int>(diag->CurrentSession())}}},
603 {"per", KvsDump()},
604 {"em", {{"running", EmRunning()}, {"processes", EmProcesses()}}},
605 {"crypto",
606#ifdef AP_NODE_CRYPTO
607 true
608#else
609 false
610#endif
611 },
612 {"logs", logs->Since(since)}};
613 if (features) j["lab"] = features->Snapshot();
614 return Json(j);
615 }
616 if (p == "/api/phm/hang" || p == "/api/phm/heal") {
617 for (auto &w : workers)
618 if (w->name == in.value("name", "")) {
619 w->hung = (p == "/api/phm/hang");
620 log->LogWarn() << (w->hung ? "fault injected: " : "fault cleared: ") << w->name;
621 return Json({{"ok", true}});
622 }
623 return Err("unknown worker", 404);
624 }
625 if (p == "/api/phm/auto-recovery") {
626 autoRecovery = in.value("enabled", true);
627 log->LogInfo() << "auto recovery " << (autoRecovery ? "enabled" : "disabled");
628 return Json({{"ok", true}});
629 }
630 if (p == "/api/system/reset") {
631 SystemReset("operator");
632 return Json({{"ok", true}});
633 }
634 if (p == "/api/diag") {
635 std::vector<std::uint8_t> r;
636 if (!FromHex(in.value("request", ""), r) || r.empty()) return Err("request must be hex bytes, e.g. 22 F1 90");
637 auto resp = diag->Dispatch(r);
638 json j = {{"ok", true}, {"request", ToHex(r)}, {"session", static_cast<int>(diag->CurrentSession())}};
639 j["response"] = resp ? json(ToHex(*resp)) : json(nullptr);
640 return Json(j);
641 }
642 if (p == "/api/per/set") {
643 std::string key = in.value("key", "");
644 if (key.empty() || !in.contains("value")) return Err("key and value required");
645 auto const &v = in["value"];
646 ara::core::Result<void> r = v.is_boolean() ? kvs->SetValue<bool>(key, v.get<bool>())
647 : v.is_number_integer() ? kvs->SetValue<std::int64_t>(key, v.get<std::int64_t>())
648 : v.is_number() ? kvs->SetValue<double>(key, v.get<double>())
649 : kvs->SetValue<std::string>(key, v.is_string() ? v.get<std::string>() : v.dump());
650 if (r.HasValue()) r = kvs->SyncToStorage();
651 if (!r.HasValue()) return Err(r.Error().Message(), 200);
652 log->LogInfo() << "persisted key '" << key << "'";
653 return Json({{"ok", true}});
654 }
655 if (p == "/api/per/delete") {
656 auto r = kvs->RemoveKey(in.value("key", ""));
657 if (r.HasValue()) r = kvs->SyncToStorage();
658 return r.HasValue() ? Json({{"ok", true}}) : Err(r.Error().Message(), 200);
659 }
660 if (p.rfind("/api/crypto/", 0) == 0) return Crypto(p.substr(12), in);
661 if (p == "/api/em/start") return Json(StartEm());
662 if (p == "/api/em/stop") {
663 StopEm();
664 return Json({{"ok", true}});
665 }
666 if (p == "/api/em/log") {
667 std::string name = req.query.count("name") ? req.query.at("name") : "EM.txt";
668 if (name.find('/') != std::string::npos || name.find("..") != std::string::npos || name.size() < 5 || name.substr(name.size() - 4) != ".txt")
669 return Err("bad file name", 403);
670 return Json({{"ok", true}, {"name", name}, {"text", ReadTail(opt.platformDir / "processes" / "redirected" / name, 40000)}});
671 }
672 if (p == "/api/em/files") {
673 json files = json::array();
674 std::error_code ec;
675 for (auto const &e : fs::directory_iterator(opt.platformDir / "processes" / "redirected", ec))
676 if (e.path().extension() == ".txt") files.push_back(e.path().filename().string());
677 return Json({{"ok", true}, {"files", files}});
678 }
679 return Err("not found", 404);
680 }
