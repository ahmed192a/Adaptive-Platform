191TEST(ComEvent, ProducerAndConsumerThreadsDoNotLoseOrCorruptSamples) {
192 ProxyEventBuffer<int> ev(256);
193 ASSERT_TRUE(ev.Subscribe(256).HasValue());
194 ev.OnSubscriptionAck(true);
195 std::thread prod([&] { for (int i = 0; i < 2000; ++i) ev.Push(i); });
196 int last = -1;
197 bool ordered = true;
198 std::size_t total = 0;
199 auto deadline = std::chrono::steady_clock::now() + std::chrono::seconds(5);
200 while ((total < 1 || prod.joinable()) && std::chrono::steady_clock::now() < deadline) {
201 auto r = ev.GetNewSamples([&](SamplePtr<int> p) { ordered &= *p > last; last = *p; });
202 total += r.Value();
203 if (last == 1999) break;
204 std::this_thread::yield();
205 }
206 prod.join();
207 ev.GetNewSamples([&](SamplePtr<int> p) { ordered &= *p > last; last = *p; });
208 EXPECT_TRUE(ordered);
209 EXPECT_EQ(last, 1999);
210}
