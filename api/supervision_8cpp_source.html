41void AliveSupervision::DoEvaluate(TimePoint now) {
42 // Close every reference cycle that has completely elapsed.
43 while (mCfg.referenceCycle > Duration::zero() && now - mCycleStart >= mCfg.referenceCycle &&
44 Status() != ElementarySupervisionStatus::kEXPIRED) {
45 std::int64_t const expected = mCfg.expectedAliveIndications;
46 std::int64_t const lo = expected - static_cast<std::int64_t>(mCfg.minMargin);
47 std::int64_t const hi = expected + static_cast<std::int64_t>(mCfg.maxMargin);
48 std::int64_t const got = mIndications;
49 if (got >= lo && got <= hi) {
50 if (mFailedCycles > 0) {
51 --mFailedCycles;
52 }
53 } else {
54 ++mFailedCycles;
55 }
56 if (mFailedCycles > mCfg.failedReferenceCyclesTolerance) {
57 Expire();
58 } else if (mFailedCycles > 0) {
59 SetStatus(ElementarySupervisionStatus::kFAILED);
60 } else {
61 SetStatus(ElementarySupervisionStatus::kOK);
62 }
63 mCycleStart += std::chrono::duration_cast<Duration>(mCfg.referenceCycle);
64 mIndications = 0;
65 }
66}

69void DeadlineSupervision::DoCheckpoint(CheckpointRef const &cp, TimePoint now) {
70 if (cp == mCfg.start) {
71 mStarted = now; // a repeated start restarts the measurement
72 } else if (cp == mCfg.end && mStarted.has_value()) {
73 Duration const elapsed = now - *mStarted;
74 mStarted.reset();
75 if (elapsed < mCfg.minDeadline || elapsed > mCfg.maxDeadline) {
76 Expire();
77 }
