71Res<ByteVector> RunKdf(char const *name, OSSL_PARAM *params, std::size_t length) {
72 std::unique_ptr<EVP_KDF, KdfDel> kdf(EVP_KDF_fetch(nullptr, name, nullptr));
73 if (!kdf) return Fail<ByteVector>(CryptoErrc::kUnsupported);
74 std::unique_ptr<EVP_KDF_CTX, KdfCtxDel> ctx(EVP_KDF_CTX_new(kdf.get()));
75 if (!ctx) return Fail<ByteVector>(CryptoErrc::kRuntimeFault);
76 ByteVector out(length);
77 if (EVP_KDF_derive(ctx.get(), out.data(), out.size(), params) != 1) {
78 SecureZero(out.data(), out.size());
79 return Fail<ByteVector>(CryptoErrc::kRuntimeFault);
80 }
81 return out;
82}

85Res<ByteVector> Hkdf(HashAlgorithm alg, ByteView ikm, ByteView salt, ByteView info, std::size_t length) {
86 if (ikm.empty()) return Fail<ByteVector>(CryptoErrc::kEmptyContainer);
87 if (length == 0 || length > 255 * DigestSize(alg)) return Fail<ByteVector>(CryptoErrc::kInvalidArgument);
88 // OpenSSL rejects a NULL data pointer even for a zero-length octet string: point empties at a dummy byte.
89 static std::uint8_t dummy = 0;
90 auto ptr = [](ByteView v) { return const_cast<std::uint8_t *>(v.empty() ? &dummy : v.data()); };
91 OSSL_PARAM params[] = {
92 OSSL_PARAM_construct_utf8_string(OSSL_KDF_PARAM_DIGEST, const_cast<char *>(MdName(alg)), 0),
93 OSSL_PARAM_construct_octet_string(OSSL_KDF_PARAM_KEY, ptr(ikm), ikm.size()),
94 OSSL_PARAM_construct_octet_string(OSSL_KDF_PARAM_SALT, ptr(salt), salt.size()),
95 OSSL_PARAM_construct_octet_string(OSSL_KDF_PARAM_INFO, ptr(info), info.size()),
96 OSSL_PARAM_construct_end()};
97 return RunKdf("HKDF", params, length);
98}

100Res<ByteVector> Pbkdf2(HashAlgorithm alg, ByteView password, ByteView salt, std::uint32_t iterations,
101 std::size_t length, bool allowWeakParameters) {
102 if (password.empty() || salt.empty()) return Fail<ByteVector>(CryptoErrc::kEmptyContainer);
103 if (iterations == 0 || length == 0) return Fail<ByteVector>(CryptoErrc::kInvalidArgument);
104 unsigned int iter = iterations;
105 int disableLowerBounds = allowWeakParameters ? 1 : 0;
106 OSSL_PARAM params[] = {
107 OSSL_PARAM_construct_utf8_string(OSSL_KDF_PARAM_DIGEST, const_cast<char *>(MdName(alg)), 0),
108 OSSL_PARAM_construct_octet_string(OSSL_KDF_PARAM_PASSWORD, const_cast<std::uint8_t *>(password.data()),
109 password.size()),
110 OSSL_PARAM_construct_octet_string(OSSL_KDF_PARAM_SALT, const_cast<std::uint8_t *>(salt.data()), salt.size()),
111 OSSL_PARAM_construct_uint(OSSL_KDF_PARAM_ITER, &iter),
112 OSSL_PARAM_construct_int(OSSL_KDF_PARAM_PKCS5, &disableLowerBounds),
113 OSSL_PARAM_construct_end()};
114 return RunKdf("PBKDF2", params, length);
115}

128Res<ByteVector> CipherRun(bool encrypt, CipherMode mode, SecretKey const &key, ByteView iv, ByteView in,
129 bool padding) {
130 std::uint32_t const need = encrypt ? kUsageEncrypt : kUsageDecrypt;
131 if ((key.Usage() & need) == 0) return Fail<ByteVector>(CryptoErrc::kUsageViolation);
132 auto const &km = KeyAccess::Bytes(key);
133 EVP_CIPHER const *c = AesCipher(mode, km.size());
134 if (c == nullptr) return Fail<ByteVector>(CryptoErrc::kInvalidKeyLength);
135 if (iv.size() != kAesBlockSize) return Fail<ByteVector>(CryptoErrc::kInvalidInputSize);
136 if (!FitsInt(in.size())) return Fail<ByteVector>(CryptoErrc::kInvalidArgument);
137 bool const pad = mode == CipherMode::kCbc && padding;
138 if (mode == CipherMode::kCbc && !pad && in.size() % kAesBlockSize != 0)
139 return Fail<ByteVector>(CryptoErrc::kInvalidInputSize);
140 if (mode == CipherMode::kCbc && !encrypt && (in.empty() || in.size() % kAesBlockSize != 0))
141 return Fail<ByteVector>(CryptoErrc::kInvalidInputSize);
142
143 CipherCtx ctx(EVP_CIPHER_CTX_new());
144 if (!ctx) return Fail<ByteVector>(CryptoErrc::kRuntimeFault);
145 if (EVP_CipherInit_ex(ctx.get(), c, nullptr, km.data(), iv.data(), encrypt ? 1 : 0) != 1)
146 return Fail<ByteVector>(CryptoErrc::kRuntimeFault);
147 EVP_CIPHER_CTX_set_padding(ctx.get(), pad ? 1 : 0);
148 ByteVector out(in.size() + kAesBlockSize);
149 int n1 = 0;
150 int n2 = 0;
151 if (in.size() > 0 &&
152 EVP_CipherUpdate(ctx.get(), out.data(), &n1, in.data(), static_cast<int>(in.size())) != 1)
153 return Fail<ByteVector>(CryptoErrc::kRuntimeFault);
154 if (EVP_CipherFinal_ex(ctx.get(), out.data() + n1, &n2) != 1) {
155 SecureZero(out.data(), out.size());
156 // padding failure on decrypt: do not expose which check failed
157 return Fail<ByteVector>(encrypt ? CryptoErrc::kRuntimeFault : CryptoErrc::kAuthTagNotValid);
158 }
159 out.resize(static_cast<std::size_t>(n1 + n2));
160 return out;
161}

183Res<ByteVector> WrapRun(bool wrap, SecretKey const &kek, ByteView in) {
184 auto const &km = KeyAccess::Bytes(kek);
185 EVP_CIPHER const *c = WrapCipher(km.size());
186 if (c == nullptr) return Fail<ByteVector>(CryptoErrc::kInvalidKeyLength);
187 if (in.size() < 16 || in.size() % 8 != 0 || !FitsInt(in.size())) return Fail<ByteVector>(CryptoErrc::kInvalidInputSize);
188 CipherCtx ctx(EVP_CIPHER_CTX_new());
189 if (!ctx) return Fail<ByteVector>(CryptoErrc::kRuntimeFault);
190 EVP_CIPHER_CTX_set_flags(ctx.get(), EVP_CIPHER_CTX_FLAG_WRAP_ALLOW);
191 if (EVP_CipherInit_ex(ctx.get(), c, nullptr, km.data(), nullptr, wrap ? 1 : 0) != 1)
192 return Fail<ByteVector>(CryptoErrc::kRuntimeFault);
193 ByteVector out(in.size() + 16);
194 int n1 = 0;
195 int n2 = 0;
196 if (EVP_CipherUpdate(ctx.get(), out.data(), &n1, in.data(), static_cast<int>(in.size())) != 1 ||
197 EVP_CipherFinal_ex(ctx.get(), out.data() + n1, &n2) != 1) {
198 SecureZero(out.data(), out.size());
199 return Fail<ByteVector>(wrap ? CryptoErrc::kRuntimeFault : CryptoErrc::kAuthTagNotValid);
200 }
201 out.resize(static_cast<std::size_t>(n1 + n2));
202 return out;
203}

243void SecretSeed::Jump(std::int64_t steps) noexcept {
244 bool const negative = steps < 0;
245 std::uint64_t mag = negative ? static_cast<std::uint64_t>(-(steps + 1)) + 1U : static_cast<std::uint64_t>(steps);
246 // add or subtract `mag` from the big-endian value, byte by byte from the least significant end
247 unsigned carry = 0;
248 for (std::size_t i = 0; i < bytes_.size(); ++i) {
249 std::uint8_t &b = bytes_[bytes_.size() - 1 - i];
250 unsigned const addend = static_cast<unsigned>(mag & 0xFFU);
251 mag >>= 8;
252 if (!negative) {
253 unsigned const sum = static_cast<unsigned>(b) + addend + carry;
254 b = static_cast<std::uint8_t>(sum & 0xFFU);
255 carry = sum >> 8;
256 } else {
257 int const diff = static_cast<int>(b) - static_cast<int>(addend) - static_cast<int>(carry);
258 b = static_cast<std::uint8_t>(diff & 0xFF);
259 carry = diff < 0 ? 1U : 0U;
260 }
261 if (mag == 0 && carry == 0) break;
262 }
263}

289Res<ByteVector> RsaSign(EVP_PKEY *key, ByteView msg) {
290 PkeyCtx ctx(EVP_PKEY_CTX_new(key, nullptr));
291 if (!ctx) return Fail<ByteVector>(CryptoErrc::kRuntimeFault);
292 // sign over the SHA-256 digest with PSS padding
293 auto d = Hash(HashAlgorithm::kSha256, msg);
294 if (!d.HasValue()) return Res<ByteVector>::FromError(d.Error());
295 if (EVP_PKEY_sign_init(ctx.get()) != 1 || EVP_PKEY_CTX_set_rsa_padding(ctx.get(), RSA_PKCS1_PSS_PADDING) != 1 ||
296 EVP_PKEY_CTX_set_rsa_pss_saltlen(ctx.get(), RSA_PSS_SALTLEN_DIGEST) != 1 ||
297 EVP_PKEY_CTX_set_signature_md(ctx.get(), EVP_sha256()) != 1)
298 return Fail<ByteVector>(CryptoErrc::kRuntimeFault);
299 std::size_t len = 0;
300 if (EVP_PKEY_sign(ctx.get(), nullptr, &len, d.Value().data(), d.Value().size()) != 1)
301 return Fail<ByteVector>(CryptoErrc::kRuntimeFault);
302 ByteVector sig(len);
303 if (EVP_PKEY_sign(ctx.get(), sig.data(), &len, d.Value().data(), d.Value().size()) != 1)
304 return Fail<ByteVector>(CryptoErrc::kRuntimeFault);
305 sig.resize(len);
306 return sig;
307}

309Res<ByteVector> RsaCrypt(bool encrypt, EVP_PKEY *key, ByteView in) {
310 PkeyCtx ctx(EVP_PKEY_CTX_new(key, nullptr));
311 if (!ctx) return Fail<ByteVector>(CryptoErrc::kRuntimeFault);
312 int const init = encrypt ? EVP_PKEY_encrypt_init(ctx.get()) : EVP_PKEY_decrypt_init(ctx.get());
313 if (init != 1 || EVP_PKEY_CTX_set_rsa_padding(ctx.get(), RSA_PKCS1_OAEP_PADDING) != 1 ||
314 EVP_PKEY_CTX_set_rsa_oaep_md(ctx.get(), EVP_sha256()) != 1 ||
315 EVP_PKEY_CTX_set_rsa_mgf1_md(ctx.get(), EVP_sha256()) != 1)
316 return Fail<ByteVector>(CryptoErrc::kRuntimeFault);
317 std::size_t len = 0;
318 auto run = [&](unsigned char *out) {
319 return encrypt ? EVP_PKEY_encrypt(ctx.get(), out, &len, in.data(), in.size())
320 : EVP_PKEY_decrypt(ctx.get(), out, &len, in.data(), in.size());
321 };
322 if (run(nullptr) != 1) return Fail<ByteVector>(encrypt ? CryptoErrc::kInvalidInputSize : CryptoErrc::kAuthTagNotValid);
323 ByteVector out(len);
324 if (run(out.data()) != 1) {
325 ERR_clear_error();
326 return Fail<ByteVector>(encrypt ? CryptoErrc::kInvalidInputSize : CryptoErrc::kAuthTagNotValid);
327 }
328 out.resize(len);
329 return out;
330}

528Res<X509Certificate> IssueCertificate(CertificateRequest const &req, X509Certificate const *issuer,
529 PrivateKey const &issuerKey) {
530 if (req.commonName.empty() || req.commonName.size() > 64 || req.subjectPublicKeyDer.empty())
531 return Fail<X509Certificate>(CryptoErrc::kInvalidArgument);
532 // rebuild the EVP keys from DER (the PrivateKey internals are private)
533 unsigned char const *sp = req.subjectPublicKeyDer.data();
534 Pkey subjectKey(d2i_PUBKEY(nullptr, &sp, static_cast<long>(req.subjectPublicKeyDer.size())));
535 if (!subjectKey) return Fail<X509Certificate>(CryptoErrc::kInvalidArgument);
536 auto pkcs8 = issuerKey.ToPkcs8Der();
537 if (!pkcs8.HasValue()) return Res<X509Certificate>::FromError(pkcs8.Error());
538 unsigned char const *kp = pkcs8.Value().data();
539 Pkey signKey(d2i_AutoPrivateKey(nullptr, &kp, static_cast<long>(pkcs8.Value().size())));
540 SecureZero(pkcs8.Value().data(), pkcs8.Value().size());
541 if (!signKey) return Fail<X509Certificate>(CryptoErrc::kRuntimeFault);
542
543 X509Ptr x(X509_new());
544 if (!x) return Fail<X509Certificate>(CryptoErrc::kRuntimeFault);
545 X509_set_version(x.get(), 2);
546 ASN1_INTEGER_set_uint64(X509_get_serialNumber(x.get()), req.serial);
547 std::time_t nb = req.notBefore != 0 ? req.notBefore : std::time(nullptr) - 60;
548 std::time_t na = req.notAfter != 0 ? req.notAfter : nb + 365L * 24 * 3600;
549 if (na <= nb) return Fail<X509Certificate>(CryptoErrc::kInvalidArgument);
550 if (X509_time_adj(X509_getm_notBefore(x.get()), 0, &nb) == nullptr ||
551 X509_time_adj(X509_getm_notAfter(x.get()), 0, &na) == nullptr)
552 return Fail<X509Certificate>(CryptoErrc::kRuntimeFault);
553 X509_NAME *subj = X509_get_subject_name(x.get());
554 if (X509_NAME_add_entry_by_txt(subj, "CN", MBSTRING_UTF8,
555 reinterpret_cast<unsigned char const *>(req.commonName.data()),
556 static_cast<int>(req.commonName.size()), -1, 0) != 1)
557 return Fail<X509Certificate>(CryptoErrc::kRuntimeFault);
558 if (issuer == nullptr) {
559 X509_set_issuer_name(x.get(), subj);
560 } else {
561 X509_set_issuer_name(x.get(), X509_get_subject_name(Raw(*issuer)));
562 }
563 X509_set_pubkey(x.get(), subjectKey.get());
564
565 X509V3_CTX v3;
566 X509V3_set_ctx_nodb(&v3);
567 X509V3_set_ctx(&v3, issuer != nullptr ? Raw(*issuer) : x.get(), x.get(), nullptr, nullptr, 0);
568 auto addExt = [&](int nid, char const *value) {
569 X509_EXTENSION *e = X509V3_EXT_conf_nid(nullptr, &v3, nid, value);
570 if (e == nullptr) return false;
571 int const rc = X509_add_ext(x.get(), e, -1);
572 X509_EXTENSION_free(e);
573 return rc == 1;
574 };
575 if (!addExt(NID_basic_constraints, req.isCa ? "critical,CA:TRUE" : "critical,CA:FALSE") ||
576 !addExt(NID_key_usage, req.isCa ? "critical,keyCertSign,cRLSign" : "critical,digitalSignature") ||
577 !addExt(NID_subject_key_identifier, "hash"))
578 return Fail<X509Certificate>(CryptoErrc::kRuntimeFault);
579 if (issuer != nullptr && !addExt(NID_authority_key_identifier, "keyid"))
580 return Fail<X509Certificate>(CryptoErrc::kRuntimeFault);
581
582 EVP_MD const *md = EVP_PKEY_base_id(signKey.get()) == EVP_PKEY_ED25519 ? nullptr : EVP_sha256();
583 if (X509_sign(x.get(), signKey.get(), md) <= 0) return Fail<X509Certificate>(CryptoErrc::kRuntimeFault);
584 return X509Certificate(std::make_shared<X509Certificate::Impl>(X509Certificate::Impl{std::move(x)}));
585}

604Res<void> X509Provider::VerifyChain(X509Certificate const &leaf, std::vector<X509Certificate> const &intermediates,
605 std::time_t at) const {
606 std::unique_ptr<X509_STORE, StoreDel> store(X509_STORE_new());
607 if (!store) return FailV(CryptoErrc::kRuntimeFault);
608 {
609 std::lock_guard<std::mutex> lk(mutex_);
610 if (anchors_.empty()) return FailV(CryptoErrc::kEmptyContainer);
611 for (auto const &a : anchors_)
612 if (X509_STORE_add_cert(store.get(), Raw(a)) != 1) return FailV(CryptoErrc::kRuntimeFault);
613 }
614 STACK_OF(X509) *inter = sk_X509_new_null();
615 if (inter == nullptr) return FailV(CryptoErrc::kRuntimeFault);
616 for (auto const &c : intermediates) {
617 X509_up_ref(Raw(c));
618 sk_X509_push(inter, Raw(c));
619 }
620 std::unique_ptr<X509_STORE_CTX, StoreCtxDel> ctx(X509_STORE_CTX_new());
621 bool ok = ctx && X509_STORE_CTX_init(ctx.get(), store.get(), Raw(leaf), inter) == 1;
622 int err = 0;
623 if (ok) {
624 X509_STORE_CTX_set_time(ctx.get(), 0, at != 0 ? at : std::time(nullptr));
625 ok = X509_verify_cert(ctx.get()) == 1;
626 if (!ok) err = X509_STORE_CTX_get_error(ctx.get());
627 }
628 ctx.reset();
629 sk_X509_pop_free(inter, X509_free);
630 ERR_clear_error();
631 if (ok) return Res<void>::FromValue();
632 return Res<void>::FromError(MakeErrorCode(CryptoErrc::kAuthTagNotValid, err));
633}
