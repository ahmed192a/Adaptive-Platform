26// ---- SIGTERM -> termination handler (SWS_EM_02577) -----------------------------------------
27// The signal handler only writes one byte to a self-pipe; a background thread runs the user handler.
28std::atomic<bool> gSlotTaken{false};
29int gPipe[2] = {-1, -1};
30std::thread gSigThread;
31std::function<void()> gTerminationHandler;
32struct sigaction gOldSigterm;
33
34void OnSigterm(int) {
35 char b = 1;
36 ssize_t const r = ::write(gPipe[1], &b, 1);
37 (void)r;
38}

40bool InstallTerminationHandler(std::function<void()> handler) {
41 if (::pipe(gPipe) != 0) return false;
42 gTerminationHandler = std::move(handler);
43 struct sigaction sa;
44 sa.sa_handler = OnSigterm;
45 sigemptyset(&sa.sa_mask);
46 sa.sa_flags = 0;
47 if (::sigaction(SIGTERM, &sa, &gOldSigterm) != 0) {
48 ::close(gPipe[0]);
49 ::close(gPipe[1]);
50 gPipe[0] = gPipe[1] = -1;
51 return false;
52 }
53 gSigThread = std::thread([] {
54 char b = 0;
55 ssize_t n;
56 do {
57 n = ::read(gPipe[0], &b, 1);
58 } while (n < 0 && errno == EINTR);
59 if (n == 1 && b == 1 && gTerminationHandler) gTerminationHandler(); // b==0: shut down by the destructor
60 });
61 return true;
62}
63
64void RemoveTerminationHandler() {
65 if (gPipe[1] >= 0) {
66 char b = 0; // wake the thread without invoking the handler
67 ssize_t const r = ::write(gPipe[1], &b, 1);
68 (void)r;
69 }
70 if (gSigThread.joinable()) gSigThread.join();
71 ::sigaction(SIGTERM, &gOldSigterm, nullptr);
72 if (gPipe[0] >= 0) ::close(gPipe[0]);
73 if (gPipe[1] >= 0) ::close(gPipe[1]);
74 gPipe[0] = gPipe[1] = -1;
75 gTerminationHandler = nullptr;
76}

79ara::core::Result<ExecutionClient> ExecutionClient::Create(std::function<void()> terminationHandler) noexcept {
80 if (!terminationHandler) {

136std::variant<std::monostate, ExecErrc> ExecutionClient::ReportExecutionState(
137 ara::exec::ExecutionState state) const noexcept {
138 std::variant<std::monostate, ExecErrc> result;
139 if (state == ara::exec::ExecutionState::kRunning &&
140 mRunningReported.load(std::memory_order_acquire)) {
141 // SWS_EM_02243: kRunning is only allowed once, from the Starting state.
142 result.emplace<ExecErrc>(ExecErrc::kInvalidTransition);
143 return result;
144 }
145 if (fd < 0 || write(fd, &state, sizeof(state)) != static_cast<ssize_t>(sizeof(state))) {
146 result.emplace<ExecErrc>(ExecErrc::kCommunicationError);
147 } else if (state == ara::exec::ExecutionState::kRunning) {
148 mRunningReported.store(true, std::memory_order_release);
149 }
150 return result;
151}
