31char const *Glyph(NmState s) {
32 switch (s) {
33 case NmState::kBusSleep: return "BusSleep ";
34 case NmState::kPrepareBusSleep: return "PrepareBusSleep";

46void Add(std::string name, std::uint8_t node, std::uint32_t pnMask) {
47 NmConfig cfg;
48 cfg.nodeId = node;
49 cfg.pnFilterMask = pnMask;
50 cfg.repeatMessageTime = Millis(600);
51 cfg.messageCycleTime = Millis(200);
52 cfg.nmTimeout = Millis(800);
53 cfg.waitBusSleepTime = Millis(500);
54 cfg.pnResetTime = Millis(1200);
55 auto r = NmChannel::Create(name, cfg, gClock);
56 if (!r.HasValue()) {
57 std::printf("cannot create %s\n", name.c_str());
58 return;
59 }
60 auto e = std::make_unique<Ecu>();
61 e->name = std::move(name);
62 e->ch = std::move(r).Value();
63 Ecu *raw = e.get();
64 raw->ch->SetStateChangeHandler([raw](NmState from, NmState to) {
65 Log(raw->name.c_str(), std::string(Glyph(from)) + " -> " + Glyph(to));
66 Millis now = gClock->Now();
67 if (raw->isAwake) raw->awake += now - raw->lastChange;
68 raw->isAwake = to != NmState::kBusSleep;
69 raw->lastChange = now;
70 });
71 raw->ch->SetTxHandler([raw](NmPdu const &p) {
72 ++gPdus;
73 for (auto &o : gEcus)
74 if (o.get() != raw) o->ch->OnPduReceived(p);
75 });
76 gEcus.push_back(std::move(e));
77}

79void Run(Millis d) {
80 for (Millis t{0}; t < d; t += Millis(10)) {
81 gClock->Advance(Millis(10));
82 for (auto &e : gEcus) e->ch->Process();
83 }
84}

109
110 std::puts("\n-- scene 3: everybody releases -> bus goes to sleep");
111 (void)Find("Camera").ch->ReleasePartialNetwork(1);

113 Run(Millis(4000));
114
115 std::puts("\n== summary ==");
116 for (auto &e : gEcus) {
117 Millis a = e->awake;
118 if (e->isAwake) a += gClock->Now() - e->lastChange;
119 std::printf("%-8s awake %5lld ms of %lld ms, final state %s\n", e->name.c_str(),
120 static_cast<long long>(a.count()), static_cast<long long>(gClock->Now().count()),
121 ara::nm::ToString(e->ch->GetState()));
122 }
123 std::printf("NM PDUs on the bus: %zu\n", gPdus);
124 return 0;
