188Res<ByteVector> MessageAuthnCode::Finish() {
189 if (!impl_) return Fail<ByteVector>(CryptoErrc::kUninitializedContext);
190 std::size_t len = 0;
191 if (EVP_DigestSignFinal(impl_->ctx.get(), nullptr, &len) != 1)
192 return Fail<ByteVector>(CryptoErrc::kRuntimeFault);
193 ByteVector tag(len);
194 if (EVP_DigestSignFinal(impl_->ctx.get(), tag.data(), &len) != 1)
195 return Fail<ByteVector>(CryptoErrc::kRuntimeFault);
196 tag.resize(len);

198 if (EVP_DigestSignInit(impl_->ctx.get(), nullptr, MdFor(impl_->alg), nullptr,
199 impl_->key.get()) != 1)
200 return Fail<ByteVector>(CryptoErrc::kRuntimeFault);

223Res<ByteVector> AeadEncrypt(SecretKey const &key, ByteView nonce, ByteView aad, ByteView pt) {
224 if ((key.usage_ & kUsageEncrypt) == 0) return Fail<ByteVector>(CryptoErrc::kUsageViolation);
225 auto const *cipher = GcmFor(key);
226 if (cipher == nullptr) return Fail<ByteVector>(CryptoErrc::kInvalidKeyLength);
227 if (nonce.size() != kGcmNonceSize || !FitsInt(pt.size()) || !FitsInt(aad.size()))
228 return Fail<ByteVector>(CryptoErrc::kInvalidArgument);
229 CipherCtx ctx(EVP_CIPHER_CTX_new());
230 if (!ctx || EVP_EncryptInit_ex(ctx.get(), cipher, nullptr, key.material_.data(), nonce.data()) != 1)
231 return Fail<ByteVector>(CryptoErrc::kRuntimeFault);
232 int n = 0;
233 if (!aad.empty() &&
234 EVP_EncryptUpdate(ctx.get(), nullptr, &n, aad.data(), static_cast<int>(aad.size())) != 1)
235 return Fail<ByteVector>(CryptoErrc::kRuntimeFault);
236 ByteVector out(pt.size() + kGcmTagSize);
237 int total = 0;
238 if (!pt.empty()) {
239 if (EVP_EncryptUpdate(ctx.get(), out.data(), &n, pt.data(), static_cast<int>(pt.size())) != 1)
240 return Fail<ByteVector>(CryptoErrc::kRuntimeFault);
241 total = n;
242 }
243 if (EVP_EncryptFinal_ex(ctx.get(), out.data() + total, &n) != 1)
244 return Fail<ByteVector>(CryptoErrc::kRuntimeFault);
245 total += n;
246 if (EVP_CIPHER_CTX_ctrl(ctx.get(), EVP_CTRL_GCM_GET_TAG, static_cast<int>(kGcmTagSize),
247 out.data() + pt.size()) != 1 ||
248 static_cast<std::size_t>(total) != pt.size())
249 return Fail<ByteVector>(CryptoErrc::kRuntimeFault);
250 return out;
251}

253Res<ByteVector> AeadDecrypt(SecretKey const &key, ByteView nonce, ByteView aad, ByteView sealed) {
254 if ((key.usage_ & kUsageDecrypt) == 0) return Fail<ByteVector>(CryptoErrc::kUsageViolation);
255 auto const *cipher = GcmFor(key);
256 if (cipher == nullptr) return Fail<ByteVector>(CryptoErrc::kInvalidKeyLength);
257 if (nonce.size() != kGcmNonceSize || !FitsInt(sealed.size()) || !FitsInt(aad.size()))
258 return Fail<ByteVector>(CryptoErrc::kInvalidArgument);
259 if (sealed.size() < kGcmTagSize) return Fail<ByteVector>(CryptoErrc::kInvalidInputSize);
260 std::size_t const ctLen = sealed.size() - kGcmTagSize;
261 CipherCtx ctx(EVP_CIPHER_CTX_new());
262 if (!ctx || EVP_DecryptInit_ex(ctx.get(), cipher, nullptr, key.material_.data(), nonce.data()) != 1)
263 return Fail<ByteVector>(CryptoErrc::kRuntimeFault);
264 int n = 0;
265 if (!aad.empty() &&
266 EVP_DecryptUpdate(ctx.get(), nullptr, &n, aad.data(), static_cast<int>(aad.size())) != 1)
267 return Fail<ByteVector>(CryptoErrc::kRuntimeFault);
268 ByteVector out(ctLen);
269 int total = 0;
270 if (ctLen != 0) {
271 if (EVP_DecryptUpdate(ctx.get(), out.data(), &n, sealed.data(), static_cast<int>(ctLen)) != 1)
272 return Fail<ByteVector>(CryptoErrc::kRuntimeFault);
273 total = n;
274 }
275 std::uint8_t tag[kGcmTagSize];
276 std::copy(sealed.data() + ctLen, sealed.data() + sealed.size(), tag);
277 if (EVP_CIPHER_CTX_ctrl(ctx.get(), EVP_CTRL_GCM_SET_TAG, static_cast<int>(kGcmTagSize), tag) != 1)
278 return Fail<ByteVector>(CryptoErrc::kRuntimeFault);
279 if (EVP_DecryptFinal_ex(ctx.get(), out.data() + total, &n) != 1) {
280 SecureZero(out.data(), out.size()); // never release unauthenticated plaintext
281 return Fail<ByteVector>(CryptoErrc::kAuthTagNotValid);
282 }
283 return out;
284}

286Res<ByteVector> AeadSeal(SecretKey const &key, ByteView aad, ByteView plaintext) {

331Res<PrivateKey> PrivateKey::FromPkcs8Der(ByteView der) {
332 if (der.empty() || !FitsInt(der.size())) return Fail<PrivateKey>(CryptoErrc::kInvalidArgument);
333 unsigned char const *p = der.data();
334 Pkey k(d2i_AutoPrivateKey(nullptr, &p, static_cast<long>(der.size())));
335 if (!k) return Fail<PrivateKey>(CryptoErrc::kInvalidArgument);
336 SignatureAlgorithm alg;
337 if (EVP_PKEY_id(k.get()) == EVP_PKEY_ED25519) {
338 alg = SignatureAlgorithm::kEd25519;
339 } else if (EVP_PKEY_id(k.get()) == EVP_PKEY_EC) {
340 char name[32] = {};

343 std::string(name, n) != "prime256v1")
344 return Fail<PrivateKey>(CryptoErrc::kUnsupported);
345 alg = SignatureAlgorithm::kEcdsaP256Sha256;
346 } else {
347 return Fail<PrivateKey>(CryptoErrc::kUnsupported);
348 }
349 return PrivateKey(std::unique_ptr<Impl>(new Impl{std::move(k)}), alg);
