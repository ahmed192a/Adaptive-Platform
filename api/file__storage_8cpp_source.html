84FileRecovery RecoverOneFile(fs::path const &storageDir, std::string const &name) {
85 auto const bak = BackupDirOf(storageDir) / name;
86 auto const crc = ReadCrc(BackupDirOf(storageDir) / (name + ".crc"));
87 auto const copy = Slurp(bak);
88 if (!crc || !copy) return FileRecovery::kNoRedundancy;
89 if (Crc32(*copy) != *crc) return FileRecovery::kRedundancyCorrupt;
90 auto const current = Slurp(storageDir / "files" / name);
91 if (current && Crc32(*current) == *crc) return FileRecovery::kIntact;
92 std::error_code ec;
93 fs::create_directories(storageDir / "files", ec);
94 if (!WriteFileAtomic(storageDir / "files" / name, *copy, false).HasValue()) {
95 return FileRecovery::kRedundancyCorrupt;
96 }
97 return FileRecovery::kRecovered;
98}

153Result<std::string> ReadAccessor::ReadRaw(std::uint64_t n) {
154 if (!mCanRead) {
155 return Result<std::string>(MakeErrorCode(PerErrc::kUnsupported, 0, "file is write-only"));
156 }
157 Position const size = GetSize();
158 if (mPosition >= size) {
159 return Result<std::string>(MakeErrorCode(PerErrc::kEof));
160 }
161 std::uint64_t const count = std::min<std::uint64_t>(n, size - mPosition);
162 std::string out(static_cast<std::size_t>(count), '\0');
163 std::size_t done = 0;
164 while (done < out.size()) {
165 ssize_t const r = ::pread(mFd, &out[done], out.size() - done,
166 static_cast<off_t>(mPosition + done));
167 if (r < 0) {
168 if (errno == EINTR) {
169 continue;
170 }
171 return Result<std::string>(detail::ErrnoToError(errno));
172 }
173 if (r == 0) {
174 break;
175 }
176 done += static_cast<std::size_t>(r);
177 }
178 out.resize(done);
179 mPosition += done;
180 return Result<std::string>(std::move(out));
181}

286Result<void> ReadWriteAccessor::SetFileSize(std::uint64_t size) {
287 if (!mCanWrite) {
288 return Result<void>(MakeErrorCode(PerErrc::kIllegalWriteAccess));
289 }
290 std::lock_guard<std::mutex> lock(mState->mutex);
291 std::uint64_t const current = GetSize();

303Result<void> ReadWriteAccessor::WriteBinary(Byte const *data, std::size_t size) {
304 if (!mCanWrite) {

306 }
307 std::lock_guard<std::mutex> lock(mState->mutex);
308 std::uint64_t const current = GetSize();

310 mPosition = current;

313 if (end > current && mState->cfg.maxSize != 0 &&

409Result<void> FileStorage::RecoverFile(StringView fileName) noexcept {
410 if (!ValidName(fileName)) {
411 return BadName();
412 }
413 if (mState->cfg.readOnly) {
414 return Result<void>(MakeErrorCode(PerErrc::kIllegalWriteAccess));
415 }
416 std::lock_guard<std::mutex> lock(mState->mutex);
417 std::string const name(fileName);

419 return Result<void>(MakeErrorCode(PerErrc::kResourceBusy));

428 return Result<void>(MakeErrorCode(PerErrc::kValidationFailed, 0, "redundancy copy is corrupt"));

432 if (!ValidName(fileName)) {
433 return BadName();

436 return Result<void>(MakeErrorCode(PerErrc::kIllegalWriteAccess));

487Result<UniqueHandle<ReadAccessor>> FileStorage::OpenFileReadOnly(StringView fileName,
488 OpenMode mode) {
489 using R = Result<UniqueHandle<ReadAccessor>>;
490 if (!ValidName(fileName)) {
491 return R(MakeErrorCode(PerErrc::kFileNotFound, 0, "invalid file name"));

514Result<UniqueHandle<ReadWriteAccessor>> FileStorage::OpenWritable(StringView fileName,
515 OpenMode mode, bool readable) {
516 using R = Result<UniqueHandle<ReadWriteAccessor>>;
517 if (!ValidName(fileName)) {
518 return R(MakeErrorCode(PerErrc::kFileNotFound, 0, "invalid file name"));
519 }
520 if (mState->cfg.readOnly) {
521 return R(MakeErrorCode(PerErrc::kIllegalWriteAccess));
522 }
523 if (Has(mode, OpenMode::kAtTheBeginning) && (Has(mode, OpenMode::kAtTheEnd) || Has(mode, OpenMode::kAppend))) {
524 return R(MakeErrorCode(PerErrc::kInvalidOpenMode));
525 }
526 std::string const name(fileName);
527 std::lock_guard<std::mutex> lock(mState->mutex);
528 if (mState->openFiles.count(name) != 0) {
529 return R(MakeErrorCode(PerErrc::kResourceBusy));
530 }
531 auto const path = mState->FilesDir() / name;
532 std::error_code ec;
533 if (!fs::exists(path, ec) && mState->cfg.maxFiles != 0) {
534 std::uint32_t count = 0;
535 for (auto const &e : fs::directory_iterator(mState->FilesDir(), ec)) {
536 count += e.is_regular_file(ec) ? 1U : 0U;
537 }
538 if (count >= mState->cfg.maxFiles) {
539 return R(MakeErrorCode(PerErrc::kTooManyFiles));
540 }
541 }
542 int flags = (readable ? O_RDWR : O_WRONLY) | O_CREAT | O_CLOEXEC;
543 if (Has(mode, OpenMode::kTruncate)) {
544 flags |= O_TRUNC;
545 }
546 int const fd = ::open(path.c_str(), flags, 0600);
547 if (fd < 0) {
548 return R(detail::ErrnoToError(errno));
549 }
550 mState->openFiles.insert(name);
551 bool const append = Has(mode, OpenMode::kAppend);
552 UniqueHandle<ReadWriteAccessor> acc(new ReadWriteAccessor(mState, name, fd, readable, true, append));
553 if (Has(mode, OpenMode::kAtTheEnd) || append) {
554 acc->mPosition = acc->GetSize();
555 }
556 return R(std::move(acc));
557}
