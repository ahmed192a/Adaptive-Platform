20void PortRegistry::Bind(std::string const &specifier, std::shared_ptr<NmChannel> const &channel) {
21 std::lock_guard<std::mutex> l(gMu);
22 auto &v = Map()[specifier];

71// SWS_ANM_00083: the aggregated network state is the lowest one (kNoCom if any channel is in No Com).
72NetworkStateType AggregateState(std::vector<std::shared_ptr<NmChannel>> const &chs) {
73 for (auto const &c : chs) {

129Result<void> NetworkHandle::RegisterNetworkStateChangeNotifier(NetworkStateChangeNotifier notifier) noexcept {
130 if (!notifier) return Result<void>::FromError(MakeErrorCode(NmErrc::kInvalidHandler));
131 auto chs = internal::PortRegistry::LookupAll(impl_->specifier);
132 if (chs.empty()) return Result<void>::FromError(MakeErrorCode(NmErrc::kServiceNotAvailable));
133 std::lock_guard<std::mutex> l(impl_->mu);
134 impl_->stateNotifier = std::move(notifier);
135 impl_->lastState = AggregateState(chs);
136 for (auto const &ch : chs) {
137 auto it = std::find_if(impl_->observed.begin(), impl_->observed.end(),
138 [&](Impl::Obs const &o) { return o.channel.lock() == ch; });
139 if (it == impl_->observed.end()) {
140 impl_->observed.push_back({ch, 0, 0});
141 it = std::prev(impl_->observed.end());
142 }
143 if (it->stateObs != 0) continue;
144 std::weak_ptr<Impl> weak = impl_;
145 // Observers run without any channel lock held; the weak_ptr makes a concurrent
146 // ~NetworkHandle safe. The notifier fires when the aggregated state changes.
147 it->stateObs = ch->AddStateObserver([weak, spec = impl_->specifier](NmState from, NmState to) {
148 if (Map(from) == Map(to)) return;
149 auto impl = weak.lock();
150 if (!impl) return;
151 auto all = internal::PortRegistry::LookupAll(spec);
152 NetworkStateType const agg = AggregateState(all);
153 NetworkStateChangeNotifier n;
154 {
155 std::lock_guard<std::mutex> g(impl->mu);
156 if (agg == impl->lastState) return;
157 impl->lastState = agg;
158 n = impl->stateNotifier;
159 }
160 if (n) n(agg);
161 });
162 }
163 return Result<void>::FromValue();
164}
