81ara::core::Result<void> Monitor::ReportMonitorAction(MonitorAction action) {
82 if (!offered_) return Err(DiagErrc::kNotOffered);
83 switch (action) {
84 case MonitorAction::kFailed:
85 counter_ = cfg_.failedThreshold;
86 server_.ReportDtc(dtc_, true);
87 break;
88 case MonitorAction::kPassed:
89 counter_ = -cfg_.passedThreshold;
90 server_.ReportDtc(dtc_, false);
91 break;
92 case MonitorAction::kPrefailed:
93 if (frozen_) break;
94 counter_ = std::max(counter_, 0) + 1;
95 if (counter_ >= cfg_.failedThreshold) {
96 counter_ = cfg_.failedThreshold;
97 server_.ReportDtc(dtc_, true);
98 }
99 break;
100 case MonitorAction::kPrepassed:
101 if (frozen_) break;
102 counter_ = std::min(counter_, 0) - 1;
103 if (-counter_ >= cfg_.passedThreshold) {
104 counter_ = -cfg_.passedThreshold;
105 server_.ReportDtc(dtc_, false);
106 }
107 break;
108 case MonitorAction::kFreezeDebouncing: frozen_ = true; break;
109 case MonitorAction::kResetDebouncing:
110 frozen_ = false;
111 counter_ = 0;
112 break;
113 default: return Err(DiagErrc::kInvalidArgument);
114 }
115 return {};
116}
