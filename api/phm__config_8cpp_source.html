24Result<PhmConfig> PhmConfig::FromJson(std::string const &text) {
25 json doc = json::parse(text, nullptr, false);
26 if (doc.is_discarded() || !doc.is_object()) {
27 return Bad("PHM config is not a JSON object");
28 }
29 PhmConfig cfg;
30 try {
31 for (auto const &s : doc.value("supervisions", json::array())) {
32 std::string const name = s.at("name").get<std::string>();
33 std::string const type = s.at("type").get<std::string>();
34 std::string const entity = s.at("entity").get<std::string>();
35 if (type == "alive") {
36 AliveSupervision::Config c;
37 c.checkpoint = {entity, s.at("checkpoint").get<CheckpointType>()};
38 c.expectedAliveIndications = s.value("expectedAliveIndications", 1U);
39 c.minMargin = s.value("minMargin", 0U);
40 c.maxMargin = s.value("maxMargin", 0U);
41 c.referenceCycle = Ms(s, "referenceCycleMs", 100);
42 c.failedReferenceCyclesTolerance = s.value("failedCyclesTolerance", 0U);
43 if (c.referenceCycle <= Duration::zero()) {
44 return Bad("referenceCycleMs must be > 0");
45 }
46 cfg.supervisions.push_back(std::make_shared<AliveSupervision>(name, c));
47 } else if (type == "deadline") {
48 DeadlineSupervision::Config c;
49 c.start = {entity, s.at("startCheckpoint").get<CheckpointType>()};
50 c.end = {entity, s.at("endCheckpoint").get<CheckpointType>()};
51 c.minDeadline = Ms(s, "minMs", 0);
52 c.maxDeadline = Ms(s, "maxMs", 100);
53 if (c.minDeadline > c.maxDeadline) {
54 return Bad("deadline minMs > maxMs");
55 }
56 cfg.supervisions.push_back(std::make_shared<DeadlineSupervision>(name, c));
57 } else if (type == "logical") {
58 LogicalSupervision::Config c;
59 c.entity = entity;
60 for (auto v : s.at("checkpoints")) c.checkpoints.insert(v.get<CheckpointType>());
61 for (auto v : s.at("initial")) c.initial.insert(v.get<CheckpointType>());
62 for (auto v : s.at("final")) c.final.insert(v.get<CheckpointType>());
63 for (auto const &t : s.at("transitions")) {
64 c.transitions.insert({t.at(0).get<CheckpointType>(), t.at(1).get<CheckpointType>()});
65 }
66 for (auto v : c.initial) {
67 if (c.checkpoints.count(v) == 0) return Bad("initial checkpoint not in 'checkpoints'");
68 }
69 cfg.supervisions.push_back(std::make_shared<LogicalSupervision>(name, c));
70 } else {
71 return Bad("supervision type must be alive, deadline or logical");
72 }
73 }
74 for (auto const &g : doc.value("globalSupervisions", json::array())) {
75 GlobalSupervisionConfig c;
76 c.name = g.at("name").get<std::string>();
77 c.members = g.at("members").get<std::vector<std::string>>();
78 c.functionGroup = g.value("functionGroup", "");
79 for (auto const &st : g.value("activeStates", json::array())) c.activeStates.insert(st.get<std::string>());
80 c.recoveryAction = g.value("recoveryAction", "");
81 c.expiredTolerance = Ms(g, "expiredToleranceMs", 0);
82 c.platformCritical = g.value("platformCritical", false);
83 c.escalateToWatchdog = g.value("escalateToWatchdog", true);
84 c.executionError = g.value("executionError", 1U);
85 cfg.globals.push_back(std::move(c));
86 }
87 } catch (json::exception const &) {
88 return Bad("missing or mistyped field in PHM config");
89 }
90 return Result<PhmConfig>(std::move(cfg));
91}
