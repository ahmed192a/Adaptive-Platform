45template <typename T, typename E> struct SharedState {
46 std::mutex mutex;
47 std::condition_variable cv;
48 std::optional<Result<T, E>> result;
49 std::vector<std::function<void()>> continuations;
50 bool futureRetrieved{false};
51
53 bool Satisfy(Result<T, E> &&r) {
54 std::vector<std::function<void()>> toRun;
55 {
56 std::lock_guard<std::mutex> lock(mutex);
57 if (result.has_value()) {
58 return false;
59 }
60 result.emplace(std::move(r));
61 toRun.swap(continuations);
62 }
63 cv.notify_all();
64 for (auto &c : toRun) {
65 c();
66 }
67 return true;
68 }
69};

154 template <typename F> auto then(F &&func) {
155 RequireState();
156 using R = std::invoke_result_t<F, Future<T, E>>;
157 using U = typename detail::UnwrapResult<R>::value;
158 auto next = std::make_shared<detail::SharedState<U, E>>();
159 auto state = std::move(mState);
160 mState.reset();
161 auto run = [state, next, fn = std::forward<F>(func)]() mutable {
162 Future<T, E> ready(state);
163 if constexpr (std::is_void_v<R>) {
164 fn(std::move(ready));
165 next->Satisfy(Result<void, E>());
166 } else if constexpr (detail::IsResult<R>::value) {
167 next->Satisfy(fn(std::move(ready)));
168 } else {
169 next->Satisfy(Result<U, E>(fn(std::move(ready))));
170 }
171 };
172 {
173 std::unique_lock<std::mutex> lock(state->mutex);
174 if (!state->result.has_value()) {
175 state->continuations.emplace_back(std::move(run));
176 return Future<U, E>(next);
177 }
178 }
179 run();
180 return Future<U, E>(next);
181 }
