38void StateClient::ErrorListener() noexcept {
39 // EM keeps the write end open, so the data survives until we attach; we poll without blocking forever.
40 int efd = -1;
41 std::string pending;
42 while (!mStop.load()) {
43 if (efd < 0) {
44 efd = ::open("execution_error_fifo", O_RDONLY | O_NONBLOCK);
45 if (efd < 0) {
46 std::this_thread::sleep_for(std::chrono::milliseconds(50));
47 continue;
48 }
49 }
50 struct pollfd p{efd, POLLIN, 0};
51 int const pr = ::poll(&p, 1, 50);
52 if (pr <= 0) continue;
53 if (p.revents & POLLIN) {
54 char buf[256];
55 ssize_t const n = ::read(efd, buf, sizeof(buf));
56 if (n > 0) pending.append(buf, static_cast<size_t>(n));
57 } else if (p.revents & (POLLHUP | POLLERR)) {
58 ::close(efd);
59 efd = -1;
60 std::this_thread::sleep_for(std::chrono::milliseconds(50));
61 }
62 for (size_t eol; (eol = pending.find('\n')) != std::string::npos;) {
63 std::string const line = pending.substr(0, eol);
64 pending.erase(0, eol + 1);
65 auto const parsed = ParseExecutionErrorLine(line);
66 if (!parsed) continue; // malformed line
67 std::string const &fg = parsed->first;
68 ExecutionError const code = parsed->second;
69 {
70 std::lock_guard<std::mutex> lock(mErrMutex);
71 mErrors[fg] = code;
72 }
73 if (mUndefinedStateCallback) {
74 FunctionGroup::CtorToken token = {{"Function_group_name", fg}, {"Mode_declarations", nlohmann::json::array()}};
75 mUndefinedStateCallback(ExecutionErrorEvent{code, FunctionGroup(std::move(token))});
76 }
77 }
78 }
79 if (efd >= 0) ::close(efd);
80}

124StateClient::SetState(FunctionGroupState const &state) const noexcept {
125 std::future<std::variant<std::monostate, ara::exec::ExecErrc>> _future;
126
127 // Messages are newline-terminated: a FIFO is a byte stream, so back-to-back requests would otherwise be
128 // read as one.
129 // The request is copied: the caller's FunctionGroupState may be gone before the task runs.
130 _future = std::async(std::launch::async, [this, msg = state.get_FGname() + "/" + state.get_states() + "\n"]() {
131 std::variant<std::monostate, ara::exec::ExecErrc> _variant = std::monostate();
132 if (fd < 0 || write(fd, msg.data(), msg.size()) != static_cast<ssize_t>(msg.size())) {
133 _variant= ara::exec::ExecErrc::kCommunicationError;
134 return _variant;
135 }
136 return _variant;
137 });
138
139 return _future;
140}
