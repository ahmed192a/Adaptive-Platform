1// supervised_service - a worker supervised by ara::phm, with a recovery action and a graceful SIGTERM exit.
2//
3// supervised_service --scenario scripted run: healthy -> hang -> EXPIRED -> recovery -> healthy (exit 0 = ok)
4// supervised_service interactive: type hang | heal | status | quit (SIGTERM also quits)
5//
6// Shows: alive supervision + global supervision + recovery action (ara::phm), the PHM_E_WATCHDOG_RESET

46void Worker() {
47 auto entity = ara::phm::SupervisedEntity::Create(ara::core::InstanceSpecifier("Svc/Worker"));
48 if (!entity.HasValue()) {
49 Say("cannot create supervised entity: ", entity.Error().Message());
50 return;
51 }
52 while (!gStop) {
53 if (!gHung) entity.Value().ReportCheckpoint(1); // "I am alive"
54 std::this_thread::sleep_for(50ms);
55 }
56}

58bool WaitFor(std::function<bool()> pred, std::chrono::milliseconds limit) {
59 auto const end = std::chrono::steady_clock::now() + limit;
60 while (std::chrono::steady_clock::now() < end) {
61 if (pred()) return true;
62 std::this_thread::sleep_for(20ms);
63 }
64 return pred();
65}

114 // Recovery action: "State Management" restarts the function group, here by un-hanging the worker.
115 std::thread restarter;
116 ara::phm::RecoveryAction recovery(
117 ara::core::InstanceSpecifier("Svc/Recovery"), [&restarter](ara::phm::ExpiredSupervision const &e) {
118 Say("[recovery] ", e.supervision, " expired in ", e.functionGroup, " (error ", e.executionError,
119 ") - restarting");
120 if (restarter.joinable()) restarter.join();
121 restarter = std::thread([fg = e.functionGroup] {
122 std::this_thread::sleep_for(300ms);
123 gHung = false;
124 ++gRestarts;
125 auto &m = ara::phm::HealthMonitor::Instance();
126 m.SetFunctionGroupState(fg, "off", ara::core::SteadyClock::now());
127 std::this_thread::sleep_for(50ms);
128 m.SetFunctionGroupState(fg, "on", ara::core::SteadyClock::now());
129 });
130 ara::core::Promise<void> p;
131 p.set_value();
132 return p.get_future();
133 });
134 if (!recovery.Offer().HasValue()) {
135 Say("cannot offer recovery action");
136 return 2;
137 }
138
139 auto const now = ara::core::SteadyClock::now();
140 mon.Start(now);
141 mon.SetFunctionGroupState("FG_Service", "on", now);
142 mon.StartBackgroundEvaluation(20ms);
143 std::thread worker(Worker);
144 Say("supervised_service running (", scenario ? "scenario" : "interactive: hang | heal | status | quit", ")");
145
146 int rc = 0;
147 if (scenario) {
148 bool ok = WaitFor(GlobalIsOk, 3s);
149 Say(ok ? "step 1/3 healthy" : "step 1/3 FAILED: never became OK");
150 if (ok) {
151 Say("step 2/3 hanging the worker");
152 gHung = true;
153 ok = WaitFor(SawExpired, 5s);
154 Say(ok ? " supervision expired" : " FAILED: no expiry");
155 }
156 if (ok) {
157 ok = WaitFor([] { return gRestarts.load() >= 1 && GlobalIsOk(); }, 8s);
158 Say(ok ? "step 3/3 recovered after restart" : "step 3/3 FAILED: no recovery");
159 }
160 rc = ok && gRestarts == 1 ? 0 : 1;
161 Say(rc == 0 ? "scenario OK" : "scenario FAILED");
162 } else {
163 std::string line;
164 while (!gStop && std::getline(std::cin, line)) {
165 if (line == "hang") {
166 gHung = true;
167 Say("worker hung");
168 } else if (line == "heal") {
169 gHung = false;
170 Say("worker released");
171 } else if (line == "status") {
172 std::lock_guard<std::mutex> lk(gMutex);
173 Say("status ", gStatus, ", restarts ", gRestarts.load());
174 } else if (line == "quit") {
175 break;
176 }
177 }
178 }
179
180 gStop = true;
181 worker.join();
182 mon.StopBackgroundEvaluation();
183 if (restarter.joinable()) restarter.join();
184 recovery.StopOffer();
185 if (exec.HasValue()) (void)exec.Value().ReportExecutionStateResult(ara::exec::ExecutionState::kTerminating);
186 return rc;
187}
