46DtcMemory DecodeDtcMemory(std::string const &text) {

50 while (std::getline(in, line)) {

53 if (!(ls >> dtcS >> stS >> occS)) continue; // malformed line: skip
54 unsigned dtc = 0, st = 0, occ = 0;

56 std::sscanf(occS.c_str(), "%u", &occ) != 1 || st > 0xFF || occ > 0xFF || dtc > 0xFFFFFF)
57 continue;
58 DtcRecord rec;
59 rec.status = static_cast<std::uint8_t>(st);
60 rec.occurrences = static_cast<std::uint8_t>(occ);
61 std::string tok;
62 while (ls >> tok) {
63 auto eq = tok.find('=');
64 if (tok.size() < 4 || eq == std::string::npos || (tok[0] != 's' && tok[0] != 'e')) continue;
65 unsigned num = 0;
66 if (std::sscanf(tok.substr(1, eq - 1).c_str(), "%u", &num) != 1 || num > 0xFF) continue;
67 ByteVector data;
68 if (!FromHex(tok.substr(eq + 1), data)) continue;

70 }
