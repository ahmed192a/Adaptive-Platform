147TEST_F(GapExec, StateClientDeliversExecutionErrorsToCallbackAndGetExecutionError) {
148 ASSERT_EQ(::mkfifo("execution_error_fifo", 0600), 0);
149 int writer = ::open("execution_error_fifo", O_RDWR | O_NONBLOCK); // EM keeps this end open
150 ASSERT_GE(writer, 0);
151 std::mutex m;
152 std::vector<std::pair<std::string, ExecutionError>> got;
153 StateClient client([&](ExecutionErrorEvent const &ev) {
154 std::lock_guard<std::mutex> lock(m);
155 got.emplace_back(ev.functionGroup.get_FGname(), ev.executionError);
156 });
157 // two events glued together + a malformed one + a split one
158 std::string const data = FormatExecutionErrorLine("FG_1", 7) + "garbage\n" + FormatExecutionErrorLine("Brake", 1);
159 ASSERT_EQ(::write(writer, data.data(), 12), 12); // "FG_1/7\ngarba"
160 std::this_thread::sleep_for(120ms);
161 ASSERT_EQ(::write(writer, data.data() + 12, data.size() - 12), static_cast<ssize_t>(data.size() - 12));
162 ASSERT_TRUE(WaitFor([&] {
163 std::lock_guard<std::mutex> lock(m);
164 return got.size() == 2;
165 }));
166 {
167 std::lock_guard<std::mutex> lock(m);
168 EXPECT_EQ(got[0], (std::pair<std::string, ExecutionError>{"FG_1", 7}));
169 EXPECT_EQ(got[1], (std::pair<std::string, ExecutionError>{"Brake", 1}));
170 }
171 FunctionGroup::CtorToken t1 = {{"Function_group_name", "FG_1"}, {"Mode_declarations", nlohmann::json::array()}};
172 FunctionGroup fg1(std::move(t1));
173 auto ev = client.GetExecutionError(fg1);
174 ASSERT_TRUE(ev.HasValue());
175 EXPECT_EQ(ev.Value().executionError, 7u);

177 FunctionGroup other(std::move(t2));
