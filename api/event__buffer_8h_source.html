113 std::size_t delivered = 0;
114 while (delivered < maxNumberOfSamples) {
115 SamplePtr<T> sp;
116 {
117 std::lock_guard<std::mutex> g(m_);
118 if (state_ != SubscriptionState::kSubscribed) break;
119 if (held_->load() >= max_)

122 if (queue_.empty()) break;
123 auto counter = held_;
124 counter->fetch_add(1);
125 sp = SamplePtr<T>(new T(std::move(queue_.front())), [counter](T const *p) {
126 delete p;
127 counter->fetch_sub(1);
128 });
129 queue_.pop_front();
130 }
131 f(std::move(sp));
132 ++delivered;
133 }

150 std::lock_guard<std::mutex> g(m_);
151 if (state_ != SubscriptionState::kSubscribed) return;
152 if (queue_.size() + held_->load() >= max_) {
153 if (queue_.empty()) return; // everything is held by the application: the new sample is lost
