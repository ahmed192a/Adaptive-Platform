21void Be32(Bytes &b, uint32_t v) {

25// Assemble a complete SD message around raw entry/option arrays.
26Bytes BuildSd(uint8_t flags, Bytes const &entries, Bytes const &options) {
27 Bytes tail;
28 tail.insert(tail.end(), {flags, 0, 0, 0});
29 Be32(tail, static_cast<uint32_t>(entries.size()));
30 tail.insert(tail.end(), entries.begin(), entries.end());
31 Be32(tail, static_cast<uint32_t>(options.size()));
32 tail.insert(tail.end(), options.begin(), options.end());
33 Bytes msg = {0xFF, 0xFF, 0x81, 0x00};
34 Be32(msg, static_cast<uint32_t>(tail.size() + 8));
35 msg.insert(msg.end(), {0x00, 0x00, 0x00, 0x01, 0x01, 0x01, 0x02, 0x00});
36 msg.insert(msg.end(), tail.begin(), tail.end());
37 return msg;
38}
